 81        """
 82        is_first = self._first_call
 83        self._first_call = False
 84
 85        payload = json.dumps(

 87                "user_input": user_input,
 88                "is_first": is_first,
 89                "continue_conversation": self.continue_conversation,
 90                "last_session_id": self._last_session_id,
 91                "setting_sources": self.setting_sources,

108        final_result = ""
109        # tool_id -> (name, start_time)

113            # print(raw_line, end="")  # Echo raw line for debugging
114            raw_line = raw_line.strip()

117

120            except json.JSONDecodeError:
121                continue
122
123            etype = event.get("type")
124
125            if etype == "session":
126                self._last_session_id = event.get("id")
127
128            elif etype == "system":
129                # print(raw_line)
130                pass
131
132            elif etype == "tool_start":
133                tool_id = event["id"]
134                tool_name = event["name"]
135                tool_input = json.dumps(event.get("input", {}))
136                # print(f"Tool '{tool_name}' started with input: {tool_input}")
137                active_tools[tool_id] = (tool_name, time.time())
138                if self.show_tool_events:
139                    self.tool_event(tool_name, tool_input, is_running=True)
140
141            elif etype == "tool_end":
142                tool_id = event["id"]
143                if tool_id in active_tools:
144                    tool_name, start = active_tools.pop(tool_id)
145                    elapsed = time.time() - start

253    def __init__(
254        self,
255        spych_object: "Spych",
256        continue_conversation: bool = True,
257        listen_duration: int | float | str = 0,
258        name: Optional[str] = None,
259        show_tool_events: bool = True,
260    ) -> None:
261        """
262        Usage:
263
264        - A responder that pipes transcribed audio into the Claude Code CLI (`claude -p`)
265          via a subprocess and returns the final response string.
266
267        Requires:
268
269        - `spych_object`:
270            - Type: Spych
271            - What: An initialized Spych instance used to record and transcribe audio
272
273        Optional:
274
275        - `continue_conversation`:
276            - Type: bool
277            - Default: True
278
279        - `listen_duration`:
280            - Type: int | float | str
281            - Default: 0
282            - Options:
283                - int | float : Record for exactly this many seconds
284                - "auto" or 0 : Use Silero VAD to detect a complete utterance and
285                                stop automatically when the speaker finishes
286
287        - `name`:
288            - Type: str
289            - Default: "Claude"
290
291        - `show_tool_events`:
292            - Type: bool
293            - Default: True
294
295        Notes:
296
297        - Uses `--output-format stream-json --verbose`
298        - The CLI may emit tool calls as raw XML inside the `result` field instead
299          of executing them via the normal agent loop. When this happens, the raw
300          result is re-submitted as a new `--resume` query on the same session,
301          mirroring the re-query loop in claude_sdk_worker.py.
302        - Intermediate assistant text (printed during tool-call turns) is printed
303          live. The final clean answer is returned to the base class for printing,
304          never printed directly, to avoid double-printing.
305        - Claude Code CLI must be installed: npm install -g @anthropic-ai/claude-code
306        """
307        name = name or "Claude"
308        super().__init__(
309            spych_object=spych_object,
310            listen_duration=listen_duration,
311            name=name,
312        )
313        self.continue_conversation = continue_conversation
314        self.show_tool_events = show_tool_events
315        self.first_call = True
316        self._last_session_id: Optional[str] = None
317
318        # Strips inline tool call XML that the CLI embeds in assistant text blocks:
319        # <function=ToolName>\n<parameter=x>val</parameter>\n</function>\n</tool_call>
320        self.TOOL_CALL_RE = re.compile(
321            r"<function=\w+>.*?(?:</function>|</tool_call>)",
322            re.DOTALL,
323        )
324
325    def __strip_tool_calls__(self, text: str) -> str:
326        """Strip inline tool-call XML from assistant text, return clean prose only."""
327        return self.TOOL_CALL_RE.sub("", text).strip()

329    def __run_turn__(
330        self,
331        user_input: str,
332        is_first: bool,
333        active_tools: dict[str, float],
334        print_assistant: bool,
335    ) -> tuple[bool, str]:
336        """
337        Run one `claude -p` subprocess turn.
338
339        Args:
340            user_input:      Prompt to send to the CLI.
341            is_first:        True only on the very first ever call (skips --resume).
342            active_tools:    Shared dict of tool_name -> start_time across turns.
343            print_assistant: If True, print intermediate assistant text live.
344                             Set False on the final turn to avoid double-printing
345                             (the base class prints the return value of respond()).
346
347        Returns:
348            (needs_continuation, result_text)
349            - needs_continuation=True means the result contained a raw
350              <tool_call> block and should be re-submitted on the same session.
351            - result_text is either the raw tool-call text (to re-submit) or
352              the clean final answer.
353        """
354        cmd = [
355            "claude",
356            "-p",
357            user_input,
358            "--output-format",
359            "stream-json",
360            "--verbose",
361        ]
362
363        if self.continue_conversation:
364            if self._last_session_id:
365                cmd.extend(["--resume", self._last_session_id])
366            elif not is_first:
367                cmd.extend(["--resume", "latest"])
368
369        proc = subprocess.Popen(
370            cmd,
371            stdin=subprocess.DEVNULL,
372            stdout=subprocess.PIPE,
373            stderr=subprocess.PIPE,
374            text=True,
375        )
376
377        result_text = ""
378
379        for raw_line in proc.stdout:
380            raw_line = raw_line.strip()
381            if not raw_line:
382                continue
383
384            try:
385                event = json.loads(raw_line)
386            except json.JSONDecodeError:
387                continue
388
389            etype = event.get("type")
390
391            if etype == "system" and event.get("subtype") == "init":
392                self._last_session_id = event.get("session_id")
393
394            elif etype == "assistant":
395                content_blocks = event.get("message", {}).get("content", [])
396                for block in content_blocks:
397                    if block.get("type") != "text":
398                        continue
399
400                    raw_text = block.get("text", "")
401
402                    # Fire a tool_start event for each new tool call seen
403                    if self.show_tool_events:
404                        for m in re.finditer(r"<function=(\w+)>", raw_text):
405                            tool_name = m.group(1)
406                            if tool_name not in active_tools:
407                                active_tools[tool_name] = time.time()
408                                preceding = raw_text[: m.start()]
409                                explanation = self.__strip_tool_calls__(
410                                    preceding
411                                ).strip()
412                                self.tool_event(
413                                    tool_name, explanation, is_running=True
414                                )
415
416                    # Only print live on intermediate (tool-call) turns.
417                    # On the final turn the base class prints the return value.
418                    if print_assistant:
419                        clean = self.__strip_tool_calls__(raw_text)
420                        if clean:
421                            self.print_response(self.name, clean)
422
423            elif etype == "result":
424                result_text = event.get("result", "")
425
426                if event.get("is_error") or event.get("subtype") == "error":
427                    for tool_name, start in list(active_tools.items()):
428                        if self.show_tool_events:
429                            self.tool_event(
430                                tool_name,
431                                "done",
432                                is_running=False,
433                                elapsed=time.time() - start,
434                            )
435                    active_tools.clear()
436                    proc.wait()
437                    return False, f"Error: {result_text}"
438
439        proc.wait()
440
441        # Mirror sdk_worker: if the result contains a raw <tool_call>, re-submit
442        if "</tool_call>" in result_text:
443            return True, result_text
444
445        # Clean final turn — close all open tool events
446        for tool_name, start in list(active_tools.items()):
447            elapsed = time.time() - start
448            if self.show_tool_events:
449                self.tool_event(
450                    tool_name, "done", is_running=False, elapsed=elapsed
451                )
452        active_tools.clear()
453
454        return False, self.__strip_tool_calls__(result_text).strip()

456    def respond(self, user_input: str) -> str:
457        """
458        Runs one or more `claude -p` subprocess turns until a clean result is
459        received. Intermediate turns (where tool calls are in flight) print
460        assistant text live. The final answer is returned for the base class
461        to print, preventing double-printing.
462        """
463        is_first = self.first_call
464        self.first_call = False
465
466        active_tools: dict[str, float] = {}
467        current_input = user_input
468
469        while True:
470            needs_continuation, result_text = self.__run_turn__(
471                current_input,
472                is_first=is_first,
473                active_tools=active_tools,
474                # Print assistant text live only on continuation turns.
475                # The final turn returns its text; the base class prints it once.
476                print_assistant=False,
477            )
478            is_first = False
479
480            if not needs_continuation:
481                return result_text
482
483            current_input = result_text

373            stdout=subprocess.PIPE,
374            stderr=subprocess.PIPE,
375            text=True,
