 60        self.frame_length = frame_length
 61        self.frame_ms = self.frame_length / self.sample_rate * 1000

 68    def record(
 69        self,
 70        device_index: int,
 71        duration: Union[int, float],
 72    ) -> list[int]:
 73        """
 74        Usage:
 75
 76        - Records audio from a microphone for a fixed duration and returns a raw
 77          PCM buffer.
 78
 79        Requires:
 80
 81        - `device_index`:
 82            - Type: int
 83            - What: The index of the microphone device to record from
 84            - Note: Use `-1` to select the system default input device
 85
 86        - `duration`:
 87            - Type: int | float
 88            - What: The number of seconds to record
 89
 90        Returns:
 91
 92        - `buffer`:
 93            - Type: list[int]
 94            - What: A flat list of raw int16 PCM samples at the configured sample rate
 95        """
 96        recorder = PvRecorder(
 97            device_index=device_index, frame_length=self.frame_length
 98        )
 99        try:
100            recorder.start()
101            frames = int(self.sample_rate * duration / self.frame_length)
102            buffer = []
103            for _ in range(frames):
104                buffer.extend(recorder.read())
105        except Exception as e:
106            raise e
107        finally:
108            recorder.stop()
109            recorder.delete()
110        return buffer

112    def record_vad(
113        self,
114        device_index: int,
115        speech_threshold: float,
116        silence_threshold: float,
117        silence_frames_threshold: int,
118        speech_pad_frames: int,
119        max_speech_duration_s: float,
120        stop_event=None,
121    ) -> list[int]:
122        """
123        Usage:
124
125        - Records from the microphone using the cached Silero VAD model to detect
126          a single complete utterance and returns the raw PCM buffer. Blocks until
127          speech is detected and then silence confirms the utterance is complete.
128
129        Requires:
130
131        - `device_index`:
132            - Type: int
133            - What: PvRecorder device index; -1 uses system default
134
135        - `speech_threshold`:
136            - Type: float
137            - What: Silero probability above which a frame is considered speech onset
138
139        - `silence_threshold`:
140            - Type: float
141            - What: Silero probability below which a frame is considered silence
142              during an active speech segment; must be less than `speech_threshold`
143              to create a hysteresis band
144
145        - `silence_frames_threshold`:
146            - Type: int
147            - What: Consecutive below-threshold frames required to confirm end of utterance
148
149        - `speech_pad_frames`:
150            - Type: int
151            - What: Pre-roll frames captured before onset confirmation; also the
152              number of consecutive voiced frames required to confirm speech onset
153
154        - `max_speech_duration_s`:
155            - Type: float
156            - What: Hard cap on utterance duration in seconds; forces a return
157              even if the speaker never pauses
158
159        Returns:
160
161        - `buffer`:
162            - Type: list[int]
163            - What: Flat list of raw int16 PCM samples representing the captured utterance
164        """
165        self.ensure_vad()
166        frame_ms = self.frame_length / self.sample_rate * 1000
167        max_speech_frames = int(max_speech_duration_s * 1000 / frame_ms)
168
169        recorder = PvRecorder(
170            device_index=device_index, frame_length=self.frame_length
171        )
172
173        speech_buffer: list[list[int]] = []
174        pre_roll: list[list[int]] = []
175        in_speech: bool = False
176        voiced_frame_count: int = 0
177        silent_frame_count: int = 0
178
179        try:
180            recorder.start()
181            while stop_event is None or not stop_event.is_set():
182                frame = recorder.read()
183                tensor = (
184                    torch.tensor(frame, dtype=torch.float32).unsqueeze(0)
185                    / 32768.0
186                )
187
188                with torch.no_grad():
189                    speech_prob = self._vad_model(
190                        tensor, self.sample_rate
191                    ).item()
192
193                if not in_speech:
194                    pre_roll.append(frame)
195                    if len(pre_roll) > speech_pad_frames:
196                        pre_roll.pop(0)
197
198                    if speech_prob >= speech_threshold:
199                        voiced_frame_count += 1
200                        if voiced_frame_count >= speech_pad_frames:
201                            in_speech = True
202                            silent_frame_count = 0
203                            speech_buffer = list(pre_roll)
204                    else:
205                        voiced_frame_count = 0
206                else:
207                    speech_buffer.append(frame)
208                    if speech_prob < silence_threshold:
209                        silent_frame_count += 1
210                    else:
211                        silent_frame_count = 0
212
213                    if (
214                        silent_frame_count >= silence_frames_threshold
215                        or len(speech_buffer) >= max_speech_frames
216                    ):
217                        break
218        finally:
219            recorder.stop()
220            recorder.delete()
221
222        return [sample for frame in speech_buffer for sample in frame]

225class Notify:
226    def notify(
227        self,
228        message: str,
229        notification_type: str = "warning",
230        depth: int = 0,
231        force: bool = False,
232    ) -> None:
233        """
234        Usage:
235
236        - Creates a class based notification message
237
238        Requires:
239
240        - `message`:
241            - Type: str
242            - What: The message to warn users with
243            - Note: Messages with `{class_name}` and `{method_name}` in them are formatted appropriately
244
245        Optional:
246
247        - `notification_type`:
248            - Type: str
249            - What: The type of notification to send (warning, verbose or exception)
250            - Default: "warning"
251            - Note:
252                - "warning" prints a warning message
253                - "verbose" prints a verbose message only if `self.verbose=True`
254                - "exception" raises an exception with the message
255
256        - `depth`:
257            - Type: int
258            - What: The depth of the nth call below the top of the method stack
259            - Note: Depth starts at 0 (indicating the current method in the stack)
260            - Default: 0
261
262        - `force`:
263            - Type: bool
264            - What: If True, forces the message to print regardless of warning or verbose settings
265            - Default: False
266
267        Notes:
268
269        - If `self.warning_stack=True`, prints the stack trace alongside warning messages
270        - If `self.warnings=False`, suppresses all warning messages
271        - If `self.verbose=True`, enables verbose messages
272        """
273        notification_types = {
274            "warning": "WARNING",
275            "verbose": "",
276            "exception": "EXCEPTION",
277        }
278        message = f"{self.__class__.__name__}.{sys._getframe(depth).f_back.f_code.co_name} {notification_types.get(notification_type, '')}: {message}"
279        if notification_type == "exception":
280            raise Exception(message)
281        elif notification_type == "warning":
282            if self.__dict__.get("warnings", True) or force:
283                if self.__dict__.get("warning_stack", False):
284                    traceback.print_stack(limit=10)
285                print(message)
286        elif notification_type == "verbose" or force:
287            if self.__dict__.get("verbose", False):
288                print(message)
289        else:
290            raise Exception(
291                f"Invalid notification type. Must be one of: {list(notification_types.keys())}"
292            )
