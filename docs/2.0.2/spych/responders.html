 38    def __init__(
 39        self,
 40        spych_object: "Spych",
 41        listen_duration: int | float = 0,
 42        name: Optional[str] = None,
 43        spinner: Optional[CliSpinner] = None,
 44        response_style: str = "",
 45        use_speaker: bool = True,
 46        speaker_voice: str = "af_heart",
 47        speaker_backend: str = "",
 48        follow_up_listen_duration: int | float = 0,
 49        inactivity_timeout: Optional[float] = 4.0,
 50        dashboard: Optional[AgentDashboard] = None,
 51        user: Optional[str] = None,
 52        allow_intermediate_responses: bool = True,
 53        display_name: Optional[str] = None,
 54    ) -> None:
 55        """
 56        Usage:
 57
 58        - Base class for all responders. Handles the listen-transcribe-respond cycle,
 59          provides a consistent interface for subclasses to implement, and includes
 60          a rich terminal UI and animated spinner.
 61
 62        - Subclasses only need to implement `respond(user_input: str) -> str`.
 63          All CLI chrome (spinner, dividers, timing, response box) is handled here.
 64
 65        - Public helper methods are available inside `respond()` for common
 66          UI needs without importing CLI internals:
 67
 68            - `self.spinner.start(message)`          — restart spinner after a pause
 69            - `self.spinner.stop()`                  — stop spinner (e.g. before printing)
 70            - `self.print_info(message, color)`      — print a styled info line
 71
 72        Requires:
 73
 74        - `spych_object`:
 75            - Type: Spych
 76            - What: An initialized Spych instance used to record and transcribe audio
 77              after the wake word is detected
 78
 79        Optional:
 80
 81        - `listen_duration`:
 82            - Type: int | float
 83            - What: The number of seconds to listen for after the wake word is detected
 84            - Default: 0 (listen indefinitely until silence is detected)
 85
 86        - `name`:
 87            - Type: str
 88            - What: A custom name for the responder to use in printed messages
 89            - Default: The class name of the responder (e.g., "Ollama")
 90
 91        - `spinner`:
 92            - Type: CliSpinner | None
 93            - What: An externally-owned spinner to share across multiple responders.
 94              When provided (e.g. by SpychOrchestrator), all responders drive the same
 95              spinner so their output never interleaves. When None, a private spinner is
 96              created, preserving the original single-responder behaviour.
 97            - Default: None
 98
 99        - `use_speaker`:
100            - Type: bool
101            - What: Whether to speak responses aloud via kokoro TTS after printing them
102            - Default: True
103
104        - `speaker_voice`:
105            - Type: str
106            - What: A kokoro voice ID used for all spoken responses
107            - Default: "af_heart"
108            - Note: American English voices use prefix `am_` or `af_`; British English
109              use `bm_` or `bf_`. See spych.speaker.Speaker for the full voice list.
110
111        - `speaker_backend`:
112            - Type: str
113            - What: Explicit TTS backend to use ("chatterbox" or "kokoro")
114            - Default: "" (priority order: Chatterbox → Kokoro)
115
116        - `response_style`:
117            - Type: str
118            - What: A style preset that shapes how the LLM formats its spoken summary.
119              Named presets: concise, friendly, military, five_year_old, fast, pirate,
120              news_anchor, haiku, shakespearean, robot, caveman, yoda, jarvis.
121              Any other string is used verbatim as a custom instruction.
122            - Default: "" (no additional style instructions)
123
124        - `follow_up_listen_duration`:
125            - Type: int | float
126            - What: How long to listen for a follow-up answer after the agent speaks
127              a response that contains a question. Uses the same duration semantics as
128              `listen_duration`: 0 uses Silero VAD to auto-detect the end of speech;
129              a positive number records for exactly that many seconds.
130            - Default: 0 (VAD auto-detect)
131            - Note: Only used when a follow-up is needed from the user.
132
133        - `inactivity_timeout`:
134            - Type: float | None
135            - What: Seconds to wait for speech onset during a listen loop
136              before pivoting back to the wake word. Only applied when using VAD-gated
137              recording (duration 0).
138            - Default: 4.0 (wait for 4 seconds of inactivity)
139
140        - `dashboard`:
141            - Type: AgentDashboard | None
142            - What: When provided, the responder routes all lifecycle events to the
143              dashboard instead of printing to stdout. The spinner is suppressed.
144            - Default: None
145
146        - `user`:
147            - Type: str | None
148            - What: The user name to use for tailored responses. If None, uses the
149              default user from settings. If "none", no user profile is used.
150            - Default: None
151
152        - `allow_intermediate_responses`:
153            - Type: bool
154            - What: When True, the LLM may return `is_intermediate_response=true` to
155              acknowledge a complex request before doing the actual work. The
156              acknowledgment is spoken async while the continuation call runs in
157              parallel. When False, the field is omitted from the schema entirely
158              and a single respond() call always returns the full answer.
159            - Default: True
160
161        - `display_name`:
162            - Type: str | None
163            - What: A human-readable name for the responder class, used specifically
164              for display in the dashboard header.
165            - Default: None (falls back to class name)
166
167        Notes:
168
169        - Subclasses must implement `respond(user_input) -> AgentResponse`.
170        - Use `self.format_prompt(prompt)` to inject the JSON schema before sending to
171          the LLM, then `self.parse_output(raw)` to convert the result to AgentResponse.
172        - When `use_speaker` is True, the `summary` field of AgentResponse is spoken in a
173          background thread so audio playback does not block the main loop.
174        - When `AgentResponse.requires_user_feedback` is True, `spoken_follow_up_loop()`
175          listens for a follow-up answer and re-runs the respond cycle inline, chaining
176          turns until no question remains or the user triggers the wake word.
177        """
178        self.spych_object = spych_object
179        self.listen_duration = listen_duration
180        self.name = name if name else self.__class__.__name__
181        self.display_name = display_name
182        self.dashboard: Optional["AgentDashboard"] = dashboard
183
184        # Set user profile
185        self.user_profile: Optional[dict] = None
186        user_name = user if user else get_default_user()
187        if user_name and user_name.lower() != "none":
188            self.user_profile = get_user(user_name)
189
190        # Use NullSpinner when a dashboard owns the screen; otherwise use a real
191        # spinner (shared if injected by the orchestrator, private otherwise).
192        _spinner: CliSpinner
193        if dashboard is not None:
194            _spinner = NullSpinner()
195        else:
196            _spinner = spinner if spinner is not None else CliSpinner()
197        self.spinner: CliSpinner = _spinner
198        self._start_time: float = 0.0
199
200        self._current_user_input: str = ""
201        self.use_speaker = use_speaker
202        self.speaker_voice = speaker_voice
203        self.speaker_backend = speaker_backend
204        self.response_style = response_style
205        self.style_hint = get_response_style(response_style)
206        self.follow_up_listen_duration = follow_up_listen_duration
207        self.inactivity_timeout = inactivity_timeout
208        self.allow_intermediate_responses = allow_intermediate_responses
209        self.speaker = None
210        self.summary_character_limit = 300
211        self._continuation_in_progress: bool = False
212        self._awaiting_follow_up: bool = False
213        self._current_turn_id: int = 0
214        self._last_speak_turn_id: int = 0
215
216        if use_speaker:
217            self.speaker = Speaker(speaker_voice, backend=speaker_backend)
218
219            def _on_playback_start():
220                if self.dashboard is not None:
221                    self.dashboard.on_status_change(
222                        "speaking", turn_id=self._last_speak_turn_id
223                    )
224
225            self.speaker.on_playback_start = _on_playback_start

386    def format_prompt(self, prompt: str) -> str:
387        """
388        Usage:
389
390        - Returns the JSON format instruction string to append to each outgoing
391          prompt. Includes a style hint when `response_style` is set.
392
393        Requires:
394
395        - `prompt`:
396            - Type: str
397            - What: The raw user input to embed at the end of the formatted prompt
398
399        Returns:
400
401        - `formatted_prompt`:
402            - Type: str
403            - What: Formatted string containing JSON schema instructions, optional style
404              hint, and the original user input; ready to send to the LLM
405        """
406        intermediate_field = (
407            """- Key `is_intermediate_response` (boolean):
408            - Set to true only when completing the user's request will require significant processing time (over 10 seconds, multiple tool calls, file edits, complex reasoning).
409            - When true, put a brief spoken acknowledgment in `response` and `summary` (e.g. "On it — this may take a moment.").
410            - If true, the calling process will present your response (or summary) to the user and then immediately tell you to continue.
411            - If you find it pertinent, you can do this multiple times as you work through very long tasks, but try to limit it to one early response right away.
412            - Steps to decide when to use this:
413                1. Do you need to call a tool, edit files, or do any complex reasoning to complete the user's request? If no, set this to false and proceed as normal.
414                2. If yes, set `is_intermediate_response` to true and provide a brief acknowledgment in `response` and `summary` to let the user know you're working on it. Don't do any actual work yet — just return this acknowledgment immediately.
415                3. After you return the intermediate response, the system will present it to the user and then tell you to continue. 
416                4. When you receive the continue signal, proceed with the actual work to complete the user's request, and return the final response with `is_intermediate_response` set to false.
417                5. In general, you should only use this once per request, right away, but in rare cases, if the task is very long and you want to provide updates, you can use it multiple times. Just make sure to only use it for significant processing time, not for quick tool calls or reasoning steps, and try to keep the user informed with a clear acknowledgment each time.
418            """
419            if self.allow_intermediate_responses
420            else ""
421        )
422        return f"""
423        You must respond with a valid json object with the following keys:
424
425        - Key: `response` (string):
426            - The full text of your response, which will be printed in the terminal.
427        - Key: `summary` (string):
428            - A short summary of your response, that will be presented at the end of each response cycle, but may be spoken outloud.
429            - Keep it casual and concise, as if you are a human summarizing your own response in a concise spoken manner.
430            - Keep as short as possible while reasonably covering the core content.
431            - Do not include any special characters or paths to files in this summary (in case it is read aloud).
432            - Ask any follow up questions at the end of this summary. Make sure to cover the core content with this summary.
433            - Do not mention any spoken instructions (like clears throat) unless to meet the instructions given by the user.
434            - Keep it less than {self.summary_character_limit} characters if possible, and definitely less than {self.summary_character_limit + 100} characters.
435        - Key: `requires_user_feedback` (boolean):
436            - A boolean flag indicating whether your response would expect a user to follow up with additional information.
437            - If true, the agent will listen for a follow-up response from the user. Make sure to ask any follow up questions last if this is true.
438        {intermediate_field}
439
440        {f"The following is information about the user you are helping. This is only for context. You should typically not reference this directly unless it is pertinent to the request stated below. START USER CONTEXT {json.dumps(self.user_profile)} END USER CONTEXT" if self.user_profile else ""}
441
442        {"Do all tasks without considering style, however, your final response should be styled like this: " if self.style_hint else ""}
443        {self.style_hint}
444
445        Everything below is the input prompt from the user:
446        =================
447
448        {prompt}
449        """

451    def parse_output(self, raw_output: str) -> AgentResponse:
452        """
453        Usage:
454
455        - Parses the raw text output from the LLM into an AgentResponse.
456          Expects the LLM to follow the format instructions provided in `format_prompt()`.
457
458        Requires:
459
460        - `raw_output`:
461            - Type: str
462            - What: The raw text output from the LLM, expected to contain a JSON object
463
464        Returns:
465
466        - `response`:
467            - Type: AgentResponse
468            - What: Structured response extracted from the LLM's JSON output; falls back
469              to echoing `raw_output` in all fields on parse failure
470        """
471        text = raw_output.strip()
472
473        # Helper for reliable boolean parsing from potentially hallucinated string booleans
474        def parse_bool(val: Any) -> bool:
475            if isinstance(val, bool):
476                return val
477            if isinstance(val, str):
478                return val.lower() == "true"
479            return bool(val)
480
481        # Extract JSON by finding the first '{' and using raw_decode to find the matching '}'.
482        # This handles nested objects and braces inside strings correctly.
483        start_idx = text.find("{")
484        if start_idx != -1:
485            try:
486                decoder = json.JSONDecoder()
487                data, _ = decoder.raw_decode(text[start_idx:])
488                return AgentResponse(
489                    response=str(data.get("response", raw_output)),
490                    summary=str(data.get("summary", raw_output)),
491                    requires_user_feedback=parse_bool(
492                        data.get("requires_user_feedback", False)
493                    ),
494                    is_intermediate_response=parse_bool(
495                        data.get("is_intermediate_response", False)
496                    ),
497                )
498            except (json.JSONDecodeError, ValueError):
499                pass
500
501        return AgentResponse(
502            response=raw_output,
503            summary=raw_output,
504            requires_user_feedback=False,
505            is_intermediate_response=False,
506        )

637    def on_user_input(
638        self, user_input: str, is_continuation: bool = False
639    ) -> None:
640        """
641        Usage:
642
643        - Print the user input and start the spinner with verbs to indicate processing.
644
645        Requires:
646
647        - `user_input`:
648            - Type: str
649            - What: The transcribed text from the user
650
651        Optional:
652
653        - `is_continuation`:
654            - Type: bool
655            - What: Whether this is a continuation turn
656            - Default: False
657
658        Notes:
659
660        - This method is called automatically when user input is received.
661        - It prints the user input label and starts the spinner with verb animation.
662        - The start time is recorded for timing purposes.
663        """
664        if not is_continuation:
665            self._current_turn_id += 1
666
667        self._current_user_input = user_input
668        if self.dashboard is not None:
669            self.dashboard.on_user_input(

770    def __call__(self) -> Optional[AgentResponse]:
771        """
772        Usage:
773
774        - Executes one full wake-triggered cycle: listens, transcribes, responds,
775          and prints.
776
777        Returns:
778
779        - `response`:
780            - Type: AgentResponse | None
781            - What: The structured response returned by `respond`, or None if
782              no user input was captured or an error occurred
783        """
784        # Interrupt any ongoing speak event when calling the responder
785        if self.speaker is not None:
786            self.speaker.interrupt()
787
788        # Set the default state for follow-up listening
789        is_follow_up = False
790
791        while True:
792            duration = (
793                self.follow_up_listen_duration
794                if is_follow_up and self.follow_up_listen_duration != 0
795                else self.listen_duration
796            )
797            inactivity_timeout = self.inactivity_timeout
798
799            self.on_listen_start(duration=duration)
800            user_input = self.spych_object.listen(
801                duration=duration, inactivity_timeout=inactivity_timeout
802            )
803            self.on_listen_end()
804
805            if not user_input:
806                self.wait_for_next_wake_word(divider=False)
807                return None
808
809            current_query = user_input
810            is_continuation_turn = False
811
812            try:
813                while True:
814                    self.on_user_input(
815                        current_query, is_continuation=is_continuation_turn
816                    )
817                    self.on_before_respond(current_query)
818                    response = self.respond(
819                        current_query, is_continuation=is_continuation_turn
820                    )
821                    self.on_after_respond(current_query, response)
822
823                    # Update internal flags immediately so callbacks see the correct state
824                    self._continuation_in_progress = bool(
825                        self.allow_intermediate_responses
826                        and response.is_intermediate_response
827                    )
828                    self._awaiting_follow_up = bool(
829                        response.requires_user_feedback
830                    )
831
832                    # Unified response handling for both intermediate and final acknowledgments.
833                    # We only print/speak if there is actual response text.
834                    if response.response:
835                        if self.speaker:
836                            self.speaker.wait_for_speak()
837                        self.on_response(response)
838
839                        if self.speaker:
840                            text_to_speak = (
841                                response.response
842                                if len(response.response)
843                                <= self.summary_character_limit
844                                else response.summary
845                            )
846                            # Capture closure variables to fix race condition in dashboard status updates (Bug 5)
847                            tid = self._current_turn_id
848                            self._last_speak_turn_id = tid
849
850                            def _on_complete():
851                                if self.dashboard is not None:
852                                    # Only the latest speech turn should update the status
853                                    if self._last_speak_turn_id != tid:
854                                        return
855
856                                    if self._continuation_in_progress:
857                                        self.dashboard.on_status_change(
858                                            "thinking", turn_id=tid
859                                        )
860                                    elif self._awaiting_follow_up:
861                                        self.dashboard.on_status_change(
862                                            "listening", turn_id=tid
863                                        )
864                                    else:
865                                        self.dashboard.on_status_change(
866                                            "waiting", turn_id=tid
867                                        )
868
869                            self.speaker.speak_async(
870                                text_to_speak, on_complete=_on_complete
871                            )
872                    else:
873                        # For final empty responses, still call on_response to show
874                        # status (e.g. failure icon) or commit the dashboard turn.
875                        # This also ensures the dashboard clears the current turn input.
876                        self.on_response(response)
877                        if self.dashboard:
878                            self.dashboard.clear_current_turn()
879
880                    if self._continuation_in_progress:
881                        is_continuation_turn = True
882                        continue
883
884                    break
885
886            except Exception as exc:
887                self._continuation_in_progress = False
888                self.spinner.stop()
889                if self.dashboard is not None:
890                    # Notify dashboard of waiting state on error and clear turn to avoid desync (Bug 1)
891                    self.dashboard.on_status_change(
892                        "waiting", turn_id=self._current_turn_id
893                    )
894                    self.dashboard.clear_current_turn()
895                else:
896                    print(f"  {theme.error}✗  Error: {exc}{theme.reset}\n")
897                if self.speaker:
898                    self.speaker.wait_for_speak()
899                self.wait_for_next_wake_word(divider=False)
900                return None
901
902            if response.requires_user_feedback:
903                if self.speaker:
904                    self._awaiting_follow_up = True
905                    self.speaker.wait_for_speak()
906                    self._awaiting_follow_up = False
907                if self.dashboard is None:
908                    CliPrinter.divider()
909                is_follow_up = True
910                # No break here; we continue the while True loop to listen for follow-up
911            else:
912                is_follow_up = False
913                self.wait_for_next_wake_word()
914                return response

443        {"Do all tasks without considering style, however, your final response should be styled like this: " if self.style_hint else ""}

452    def parse_output(self, raw_output: str) -> AgentResponse:
453        """
454        Usage:
455
456        - Parses the raw text output from the LLM into an AgentResponse.
457          Expects the LLM to follow the format instructions provided in `format_prompt()`.
458
459        Requires:
460
461        - `raw_output`:
462            - Type: str
463            - What: The raw text output from the LLM, expected to contain a JSON object
464
465        Returns:
466
467        - `response`:
468            - Type: AgentResponse
469            - What: Structured response extracted from the LLM's JSON output; falls back
470              to echoing `raw_output` in all fields on parse failure
471        """
472        text = raw_output.strip()
473
474        # Helper for reliable boolean parsing from potentially hallucinated string booleans
475        def parse_bool(val: Any) -> bool:
476            if isinstance(val, bool):
477                return val
478            if isinstance(val, str):
479                return val.lower() == "true"
480            return bool(val)
481
482        # Extract JSON by finding the first '{' and using raw_decode to find the matching '}'.
483        # This handles nested objects and braces inside strings correctly.
484        start_idx = text.find("{")
485        if start_idx != -1:
486            try:
487                decoder = json.JSONDecoder()
488                data, _ = decoder.raw_decode(text[start_idx:])
489                return AgentResponse(
490                    response=str(data.get("response", raw_output)),
491                    summary=str(data.get("summary", raw_output)),
492                    requires_user_feedback=parse_bool(
493                        data.get("requires_user_feedback", False)
494                    ),
495                    is_intermediate_response=parse_bool(
496                        data.get("is_intermediate_response", False)
497                    ),
498                )
499            except (json.JSONDecodeError, ValueError):
500                pass
501
502        return AgentResponse(
503            response=raw_output,
504            summary=raw_output,
505            requires_user_feedback=False,
506            is_intermediate_response=False,
507        )

771    def __call__(self) -> Optional[AgentResponse]:
772        """
773        Usage:
774
775        - Executes one full wake-triggered cycle: listens, transcribes, responds,
776          and prints.
777
778        Returns:
779
780        - `response`:
781            - Type: AgentResponse | None
782            - What: The structured response returned by `respond`, or None if
783              no user input was captured or an error occurred
784        """
785        # Interrupt any ongoing speak event when calling the responder
786        if self.speaker is not None:
787            self.speaker.interrupt()
788
789        # Set the default state for follow-up listening
790        is_follow_up = False
791
792        while True:
793            duration = (
794                self.follow_up_listen_duration
795                if is_follow_up and self.follow_up_listen_duration != 0
796                else self.listen_duration
797            )
798            inactivity_timeout = self.inactivity_timeout
799
800            self.on_listen_start(duration=duration)
801            user_input = self.spych_object.listen(
802                duration=duration, inactivity_timeout=inactivity_timeout
803            )
804            self.on_listen_end()
805
806            if not user_input:
807                self.wait_for_next_wake_word(divider=False)
808                return None
809
810            current_query = user_input
811            is_continuation_turn = False
812
813            try:
814                while True:
815                    self.on_user_input(
816                        current_query, is_continuation=is_continuation_turn
817                    )
818                    self.on_before_respond(current_query)
819                    response = self.respond(
820                        current_query, is_continuation=is_continuation_turn
821                    )
822                    self.on_after_respond(current_query, response)
823
824                    # Update internal flags immediately so callbacks see the correct state
825                    self._continuation_in_progress = bool(
826                        self.allow_intermediate_responses
827                        and response.is_intermediate_response
828                    )
829                    self._awaiting_follow_up = bool(
830                        response.requires_user_feedback
831                    )
832
833                    # Unified response handling for both intermediate and final acknowledgments.
834                    # We only print/speak if there is actual response text.
835                    if response.response:
836                        if self.speaker:
837                            self.speaker.wait_for_speak()
838                        self.on_response(response)
839
840                        if self.speaker:
841                            text_to_speak = (
842                                response.response
843                                if len(response.response)
844                                <= self.summary_character_limit
845                                else response.summary
846                            )
847                            # Capture closure variables to fix race condition in dashboard status updates (Bug 5)
848                            tid = self._current_turn_id
849                            self._last_speak_turn_id = tid
850
851                            def _on_complete():
852                                if self.dashboard is not None:
853                                    # Only the latest speech turn should update the status
854                                    if self._last_speak_turn_id != tid:
855                                        return
856
857                                    if self._continuation_in_progress:
858                                        self.dashboard.on_status_change(
859                                            "thinking", turn_id=tid
860                                        )
861                                    elif self._awaiting_follow_up:
862                                        self.dashboard.on_status_change(
863                                            "listening", turn_id=tid
864                                        )
865                                    else:
866                                        self.dashboard.on_status_change(
867                                            "waiting", turn_id=tid
868                                        )
869
870                            self.speaker.speak_async(
871                                text_to_speak, on_complete=_on_complete
872                            )
873                    else:
874                        # For final empty responses, still call on_response to show
875                        # status (e.g. failure icon) or commit the dashboard turn.
876                        # This also ensures the dashboard clears the current turn input.
877                        self.on_response(response)
878                        if self.dashboard:
879                            self.dashboard.clear_current_turn()
880
881                    if self._continuation_in_progress:
882                        is_continuation_turn = True
883                        continue
884
885                    break
886
887            except Exception as exc:
888                self._continuation_in_progress = False
889                self.spinner.stop()
890                if self.dashboard is not None:
891                    # Notify dashboard of waiting state on error and clear turn to avoid desync (Bug 1)
892                    self.dashboard.on_status_change(
893                        "waiting", turn_id=self._current_turn_id
894                    )
895                    self.dashboard.clear_current_turn()
896                else:
897                    print(f"  {theme.error}✗  Error: {exc}{theme.reset}\n")
898                if self.speaker:
899                    self.speaker.wait_for_speak()
900                self.wait_for_next_wake_word(divider=False)
901                return None
902
903            if response.requires_user_feedback:
904                if self.speaker:
905                    self._awaiting_follow_up = True
906                    self.speaker.wait_for_speak()
907                    self._awaiting_follow_up = False
908                if self.dashboard is None:
909                    CliPrinter.divider()
910                is_follow_up = True
911                # No break here; we continue the while True loop to listen for follow-up
912            else:
913                is_follow_up = False
914                self.wait_for_next_wake_word()
915                return response

453        """
454        Usage:

458
459        Requires:

464

468            - Type: AgentResponse

470              to echoing `raw_output` in all fields on parse failure
471        """
472        text = raw_output.strip()
473
474        # Helper for reliable boolean parsing from potentially hallucinated string booleans
475        def parse_bool(val: Any) -> bool:
476            if isinstance(val, bool):
477                return val
478            if isinstance(val, str):
479                return val.lower() == "true"
480            return bool(val)
481
482        # Extract JSON by finding the first '{' and using raw_decode to find the matching '}'.
483        # This handles nested objects and braces inside strings correctly.
484        start_idx = text.find("{")
485        if start_idx != -1:
486            try:
487                decoder = json.JSONDecoder()
488                data, _ = decoder.raw_decode(text[start_idx:])
489                return AgentResponse(
490                    response=str(data.get("response", raw_output)),
491                    summary=str(data.get("summary", raw_output)),

493                        data.get("requires_user_feedback", False)

497                    ),
498                )
