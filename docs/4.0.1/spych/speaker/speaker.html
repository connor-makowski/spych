 15    def __init__(self, voice: str = "", backend: str = "") -> None:
 16        """
 17        Usage:
 18
 19        - Initializes a Speaker that converts text to speech.
 20          Attempts to use Chatterbox (high quality, slower) first.
 21          Falls back to Kokoro (lightweight) if Chatterbox is not available.
 22          Falls back to a silent (print-only) backend if neither is available.
 23
 24        Optional:
 25
 26        - `voice`:
 27            - Type: str
 28            - What: Voice name used by the active backend.
 29              For Chatterbox: a wave voice name (e.g. "af_heart") or a path to a
 30              .wav file. Wave voices are downloaded from the spych repo on first use.
 31              For Kokoro: a Kokoro voice name (e.g. "af_heart", "bm_george").
 32              The same short names (af_heart, bm_george, etc.) work for both backends.
 33            - Default: "" — Chatterbox requires a voice and will raise ValueError if
 34              empty; Kokoro defaults to "af_heart".
 35
 36        - `backend`:
 37            - Type: str
 38            - What: Explicit backend to use ("chatterbox" or "kokoro").
 39            - Default: "" (priority order: Chatterbox → Kokoro)
 40        """
 41        pygame.mixer.init()
 42        self.voice = voice
 43        self.interrupted = threading.Event()
 44        self.speaking_complete = threading.Event()
 45        self.speaking_complete.set()
 46
 47        self.backend = get_backend(
 48            speaker=self, voice=voice, backend_name=backend
 49        )
 50
 51    def play_pcm_array(self, audio: np.ndarray, sample_rate: int) -> None:
 52        """Helper to play a numpy PCM array via pygame."""
 53        pcm = (audio.flatten() * 32767).astype(np.int16)
 54        buf = io.BytesIO()
 55        with wave.open(buf, "wb") as wf:
 56            wf.setnchannels(1)
 57            wf.setsampwidth(2)
 58            wf.setframerate(sample_rate)
 59            wf.writeframes(pcm.tobytes())
 60
 61        buf.seek(0)
 62        pygame.mixer.music.load(buf)
 63        pygame.mixer.music.play()
 64
 65        while pygame.mixer.music.get_busy():
 66            if self.interrupted.is_set():
 67                pygame.mixer.music.stop()
 68                return
 69            time.sleep(0.05)
 70
 71    def speak(self, text: str) -> None:
 72        """
 73        Usage:
 74
 75        - Converts text to speech and plays it through the system audio output.
 76          Blocks until playback is complete.
 77        """
 78        self.interrupted.clear()
 79        self.backend.speak(text)
 80
 81    def speak_async(self, text: str) -> None:
 82        """
 83        Usage:
 84
 85        - Converts text to speech and plays it through the system audio output in
 86          a background thread. Does not block the main thread.
 87        """
 88
 89        def run_speak():
 90            try:
 91                if not self.interrupted.is_set():
 92                    self.speak(text)
 93            except Exception:
 94                pass
 95            finally:
 96                self.speaking_complete.set()
 97
 98        self.speaking_complete.clear()
 99        self.interrupted.clear()
100        threading.Thread(target=run_speak, daemon=True).start()
101
102    def wait_for_speak(self) -> None:
103        """Blocks until the current speak call is complete."""
104        self.speaking_complete.wait()
105
106    def is_speaking(self) -> bool:
107        """Returns True if a speak call is in progress."""
108        return not self.speaking_complete.is_set()
109
110    def interrupt(self) -> None:
111        """Stops any in-progress speak call immediately."""
112        self.interrupted.set()
113        pygame.mixer.music.stop()
