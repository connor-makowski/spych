  9def resolve_cmd(name: str) -> str:
 10    """
 11    Usage:
 12
 13    - Resolves a CLI command name to its full executable path using shutil.which.
 14      On Windows, Node.js CLI tools are installed as .cmd wrappers (e.g. gemini.cmd).
 15      subprocess.Popen with shell=False cannot find them by bare name, but shutil.which
 16      respects PATHEXT and returns the full path including the extension.
 17
 18    Requires:
 19
 20    - `name`:
 21        - Type: str
 22        - What: The CLI command name to resolve (e.g. "gemini", "claude")
 23
 24    Returns:
 25
 26    - `path`:
 27        - Type: str
 28        - What: The full path to the executable if found, otherwise the original name
 29    """
 30    return shutil.which(name) or name

 33def get_cache_dir(folder="voices") -> str:
 34    """Returns the path to the project's voice cache directory."""
 35    if os.name == "nt":  # Windows
 36        base_dir = os.environ.get("LOCALAPPDATA", os.path.expanduser("~"))
 37    elif sys.platform == "darwin":  # macOS
 38        base_dir = os.path.expanduser("~/Library/Caches")
 39    else:  # Linux/Unix
 40        base_dir = os.environ.get(
 41            "XDG_CACHE_HOME", os.path.expanduser("~/.cache")
 42        )
 43
 44    path = os.path.join(base_dir, "spych", folder)
 45    os.makedirs(path, exist_ok=True)
 46    return path

 49def save_wav(path: str, buffer: list[int], sample_rate: int = 16000) -> None:
 50    """Saves a raw PCM buffer to a .wav file."""
 51    with wave.open(path, "wb") as wf:
 52        wf.setnchannels(1)
 53        wf.setsampwidth(2)
 54        wf.setframerate(sample_rate)
 55        wf.writeframes(np.array(buffer, dtype=np.int16).tobytes())

 58def get_clean_audio_buffer(buffer: list[int]) -> np.ndarray:
 59    """
 60    Usage:
 61
 62    - Converts a raw int16 PCM buffer into a normalized float32 numpy array
 63      suitable for use with faster-whisper
 64
 65    Requires:
 66
 67    - `buffer`:
 68        - Type: list[int]
 69        - What: A flat list of raw int16 PCM samples, as returned by `Recorder.record`
 70
 71    Returns:
 72
 73    - `audio_buffer`:
 74        - Type: np.ndarray
 75        - What: A float32 numpy array with values normalized to the range [-1.0, 1.0]

 77    """
 78    return np.array(buffer, dtype=np.int16).astype(np.float32) / 32768.0

 81class Recorder:
 82    def __init__(
 83        self,
 84        sample_rate: int = 16000,
 85        frame_length: int = 512,
 86    ) -> None:
 87        """
 88        Usage:
 89
 90        - Initializes a recorder that wraps PvRecorder for fixed-duration recording
 91          and Silero VAD for utterance-gated recording. The VAD model is loaded once
 92          at first call and reused across all `record_vad` calls.
 93
 94        Optional:
 95
 96        - `sample_rate`:
 97            - Type: int
 98            - What: The sample rate in Hz used for both recording and VAD inference
 99            - Default: 16000
100            - Note: Should match the sample rate expected by your downstream model
101
102        - `frame_length`:
103            - Type: int
104            - What: The number of samples per frame read from PvRecorder
105            - Default: 512
106            - Note: Also used as Silero VAD's native 16kHz window (~32ms per frame)
107        """
108        self.sample_rate = sample_rate
109        self.frame_length = frame_length
110        self.frame_ms = self.frame_length / self.sample_rate * 1000
111
112    def ensure_vad(self) -> None:
113        if not hasattr(self, "_vad_model"):
114            self._vad_model = load_silero_vad()
115            self._vad_model.eval()

117    def record(
118        self,
119        device_index: int,
120        duration: Union[int, float],
121    ) -> list[int]:
122        """
123        Usage:
124
125        - Records audio from a microphone for a fixed duration and returns a raw
126          PCM buffer.
127
128        Requires:
129
130        - `device_index`:
131            - Type: int
132            - What: The index of the microphone device to record from
133            - Note: Use `-1` to select the system default input device
134
135        - `duration`:
136            - Type: int | float
137            - What: The number of seconds to record
138
139        Returns:
140
141        - `buffer`:
142            - Type: list[int]
143            - What: A flat list of raw int16 PCM samples at the configured sample rate
144        """
145        recorder = PvRecorder(
146            device_index=device_index, frame_length=self.frame_length
147        )
148        try:
149            recorder.start()
150            frames = int(self.sample_rate * duration / self.frame_length)
151            buffer = []
152            for _ in range(frames):
153                buffer.extend(recorder.read())
154        except Exception as e:
155            raise e
156        finally:
157            recorder.stop()
158            recorder.delete()
159        return buffer

161    def record_vad(
162        self,
163        device_index: int,
164        speech_threshold: float,
165        silence_threshold: float,
166        silence_frames_threshold: int,
167        speech_pad_frames: int,
168        max_speech_duration_s: float,
169        inactivity_timeout: Optional[float] = None,
170        stop_event=None,
171    ) -> list[int]:
172        """
173        Usage:
174
175        - Records from the microphone using the cached Silero VAD model to detect
176          a single complete utterance and returns the raw PCM buffer. Blocks until
177          speech is detected and then silence confirms the utterance is complete.
178
179        Requires:
180
181        - `device_index`:
182            - Type: int
183            - What: PvRecorder device index; -1 uses system default
184
185        - `speech_threshold`:
186            - Type: float
187            - What: Silero probability above which a frame is considered speech onset
188
189        - `silence_threshold`:
190            - Type: float
191            - What: Silero probability below which a frame is considered silence
192              during an active speech segment; must be less than `speech_threshold`
193              to create a hysteresis band
194
195        - `silence_frames_threshold`:
196            - Type: int
197            - What: Consecutive below-threshold frames required to confirm end of utterance
198
199        - `speech_pad_frames`:
200            - Type: int
201            - What: Pre-roll frames captured before onset confirmation; also the
202              number of consecutive voiced frames required to confirm speech onset
203
204        - `max_speech_duration_s`:
205            - Type: float
206            - What: Hard cap on utterance duration in seconds; forces a return
207              even if the speaker never pauses
208
209        Optional:
210
211        - `inactivity_timeout`:
212            - Type: float | None
213            - What: Seconds to wait for speech onset before returning an empty buffer
214            - Default: None (wait indefinitely)
215
216        Returns:
217
218        - `buffer`:
219            - Type: list[int]
220            - What: Flat list of raw int16 PCM samples representing the captured utterance
221        """
222        self.ensure_vad()
223        frame_ms = self.frame_length / self.sample_rate * 1000
224        max_speech_frames = int(max_speech_duration_s * 1000 / frame_ms)
225        inactivity_frames = (
226            int(inactivity_timeout * 1000 / frame_ms)
227            if inactivity_timeout
228            else None
229        )
230
231        recorder = PvRecorder(
232            device_index=device_index, frame_length=self.frame_length
233        )
234
235        speech_buffer: list[list[int]] = []
236        pre_roll: list[list[int]] = []
237        in_speech: bool = False
238        voiced_frame_count: int = 0
239        silent_frame_count: int = 0
240        total_frames: int = 0
241
242        try:
243            recorder.start()
244            while stop_event is None or not stop_event.is_set():
245                frame = recorder.read()
246                total_frames += 1
247                tensor = (
248                    torch.tensor(frame, dtype=torch.float32).unsqueeze(0)
249                    / 32768.0
250                )
251
252                with torch.no_grad():
253                    speech_prob = self._vad_model(
254                        tensor, self.sample_rate
255                    ).item()
256
257                if not in_speech:
258                    if (
259                        inactivity_frames is not None
260                        and total_frames >= inactivity_frames
261                    ):
262                        break
263
264                    pre_roll.append(frame)
265                    if len(pre_roll) > speech_pad_frames:
266                        pre_roll.pop(0)
267
268                    if speech_prob >= speech_threshold:
269                        voiced_frame_count += 1
270                        if voiced_frame_count >= speech_pad_frames:
271                            in_speech = True
272                            silent_frame_count = 0
273                            speech_buffer = list(pre_roll)
274                    else:
275                        voiced_frame_count = 0
276                else:
277                    speech_buffer.append(frame)
278                    if speech_prob < silence_threshold:
279                        silent_frame_count += 1
280                    else:
281                        silent_frame_count = 0
282
283                    if (
284                        silent_frame_count >= silence_frames_threshold
285                        or len(speech_buffer) >= max_speech_frames
286                    ):
287                        break
288        finally:
289            recorder.stop()
290            recorder.delete()
291
292        return [sample for frame in speech_buffer for sample in frame]

295class Notify:
296    def notify(
297        self,
298        message: str,
299        notification_type: str = "warning",
300        depth: int = 0,
301        force: bool = False,
302    ) -> None:
303        """
304        Usage:
305
306        - Creates a class based notification message
307
308        Requires:
309
310        - `message`:
311            - Type: str
312            - What: The message to warn users with
313            - Note: Messages with `{class_name}` and `{method_name}` in them are formatted appropriately
314
315        Optional:
316
317        - `notification_type`:
318            - Type: str
319            - What: The type of notification to send (warning, verbose or exception)
320            - Default: "warning"
321            - Note:
322                - "warning" prints a warning message
323                - "verbose" prints a verbose message only if `self.verbose=True`
324                - "exception" raises an exception with the message
325
326        - `depth`:
327            - Type: int
328            - What: The depth of the nth call below the top of the method stack
329            - Note: Depth starts at 0 (indicating the current method in the stack)
330            - Default: 0
331
332        - `force`:
333            - Type: bool
334            - What: If True, forces the message to print regardless of warning or verbose settings
335            - Default: False
336
337        Notes:
338
339        - If `self.warning_stack=True`, prints the stack trace alongside warning messages
340        - If `self.warnings=False`, suppresses all warning messages
341        - If `self.verbose=True`, enables verbose messages
342        """
343        notification_types = {
344            "warning": "WARNING",
345            "verbose": "",
346            "exception": "EXCEPTION",
347        }
348        message = f"{self.__class__.__name__}.{sys._getframe(depth).f_back.f_code.co_name} {notification_types.get(notification_type, '')}: {message}"
349        if notification_type == "exception":
350            raise Exception(message)
351        elif notification_type == "warning":
352            if self.__dict__.get("warnings", True) or force:
353                if self.__dict__.get("warning_stack", False):
354                    traceback.print_stack(limit=10)
355                print(message)
356        elif notification_type == "verbose" or force:
357            if self.__dict__.get("verbose", False):
358                print(message)
359        else:
360            raise Exception(
361                f"Invalid notification type. Must be one of: {list(notification_types.keys())}"
362            )

465def get_personality(name: str) -> dict:
466    """
467    Usage:
468
469    - Returns a personality preset dict containing default kwargs (name,
470      wake_words, speaker_voice, use_speaker, response_style) for the named
471      personality. Raises ValueError for unknown names.
472
473    Requires:
474
475    - `name`:
476        - Type: str
477        - What: The personality preset name (e.g. "jarvis")
478
479    Returns:
480
481    - `preset`:
482        - Type: dict
483        - What: A dict of agent kwargs to apply as defaults.
484
485    Notes:
486
487    - Raises ValueError if the name is not found in PERSONALITIES.
488    """
489    key = name.lower()
490    if key not in PERSONALITIES:
491        valid = ", ".join(sorted(PERSONALITIES))
492        raise ValueError(f"Unknown personality {name!r}. Valid options: {valid}")
493    return dict(PERSONALITIES[key])
