 10def resolve_cmd(name: str) -> str:
 11    """
 12    Usage:
 13
 14    - Resolves a CLI command name to its full executable path using shutil.which.
 15      On Windows, Node.js CLI tools are installed as .cmd wrappers (e.g. gemini.cmd).
 16      While shutil.which returns the full path including the extension,
 17      subprocess.Popen on Windows often requires shell=True to correctly
 18      execute these batch wrappers.
 19
 20    Requires:
 21
 22    - `name`:
 23        - Type: str
 24        - What: The CLI command name to resolve (e.g. "gemini", "claude")
 25
 26    Returns:
 27
 28    - `path`:
 29        - Type: str
 30        - What: The full path to the executable if found, otherwise the original name
 31    """
 32    return shutil.which(name) or name

 35def supports_unicode() -> bool:
 36    """
 37    Check if the terminal likely supports Unicode icons (like braille or emoji).
 38    On Windows, we check for 'utf-8' encoding or typical modern terminal indicators.
 39    """
 40    if sys.platform != "win32":
 41        return True
 42    # Modern Windows Terminal or VS Code terminal usually support Unicode
 43    if os.environ.get("WT_SESSION") or os.environ.get("VSCODE_PID"):
 44        return True
 45    # If the output encoding is explicitly set to utf-8, we trust it
 46    if getattr(sys.stdout, "encoding", "").lower() == "utf-8":
 47        return True
 48    return False

 51def get_cache_dir(folder="voices") -> str:
 52    """Returns the path to the project's voice cache directory."""
 53    if os.name == "nt":  # Windows
 54        base_dir = os.environ.get("LOCALAPPDATA", os.path.expanduser("~"))
 55    elif sys.platform == "darwin":  # macOS
 56        base_dir = os.path.expanduser("~/Library/Caches")
 57    else:  # Linux/Unix
 58        base_dir = os.environ.get(
 59            "XDG_CACHE_HOME", os.path.expanduser("~/.cache")
 60        )
 61
 62    path = os.path.join(base_dir, "spych", folder)
 63    os.makedirs(path, exist_ok=True)
 64    return path

164class Recorder:
165    def __init__(
166        self,
167        sample_rate: int = 16000,
168        frame_length: int = 512,
169    ) -> None:
170        """
171        Usage:
172
173        - Initializes a recorder that wraps PvRecorder for fixed-duration recording
174          and Silero VAD for utterance-gated recording. The VAD model is loaded once
175          at first call and reused across all `record_vad` calls.
176
177        Optional:
178
179        - `sample_rate`:
180            - Type: int
181            - What: The sample rate in Hz used for both recording and VAD inference
182            - Default: 16000
183            - Note: Should match the sample rate expected by your downstream model
184
185        - `frame_length`:
186            - Type: int
187            - What: The number of samples per frame read from PvRecorder
188            - Default: 512
189            - Note: Also used as Silero VAD's native 16kHz window (~32ms per frame)
190        """
191        self.sample_rate = sample_rate
192        self.frame_length = frame_length
193        self.frame_ms = self.frame_length / self.sample_rate * 1000
194
195    def ensure_vad(self) -> None:
196        if not hasattr(self, "_vad_model"):
197            self._vad_model = load_silero_vad()
198            self._vad_model.eval()

200    def record(
201        self,
202        device_index: int,
203        duration: Union[int, float],
204    ) -> list[int]:
205        """
206        Usage:
207
208        - Records audio from a microphone for a fixed duration and returns a raw
209          PCM buffer.
210
211        Requires:
212
213        - `device_index`:
214            - Type: int
215            - What: The index of the microphone device to record from
216            - Note: Use `-1` to select the system default input device
217
218        - `duration`:
219            - Type: int | float
220            - What: The number of seconds to record
221
222        Returns:
223
224        - `buffer`:
225            - Type: list[int]
226            - What: A flat list of raw int16 PCM samples at the configured sample rate
227        """
228        recorder = PvRecorder(
229            device_index=device_index, frame_length=self.frame_length
230        )
231        try:
232            recorder.start()
233            frames = int(self.sample_rate * duration / self.frame_length)
234            buffer = []
235            for _ in range(frames):
236                buffer.extend(recorder.read())
237        except Exception as e:
238            raise e
239        finally:
240            recorder.stop()
241            recorder.delete()
242        return buffer

244    def record_vad(
245        self,
246        device_index: int,
247        speech_threshold: float,
248        silence_threshold: float,
249        silence_frames_threshold: int,
250        speech_pad_frames: int,
251        max_speech_duration_s: float,
252        inactivity_timeout: Optional[float] = None,
253        stop_event=None,
254    ) -> list[int]:
255        """
256        Usage:
257
258        - Records from the microphone using the cached Silero VAD model to detect
259          a single complete utterance and returns the raw PCM buffer. Blocks until
260          speech is detected and then silence confirms the utterance is complete.
261
262        Requires:
263
264        - `device_index`:
265            - Type: int
266            - What: PvRecorder device index; -1 uses system default
267
268        - `speech_threshold`:
269            - Type: float
270            - What: Silero probability above which a frame is considered speech onset
271
272        - `silence_threshold`:
273            - Type: float
274            - What: Silero probability below which a frame is considered silence
275              during an active speech segment; must be less than `speech_threshold`
276              to create a hysteresis band
277
278        - `silence_frames_threshold`:
279            - Type: int
280            - What: Consecutive below-threshold frames required to confirm end of utterance
281
282        - `speech_pad_frames`:
283            - Type: int
284            - What: Pre-roll frames captured before onset confirmation; also the
285              number of consecutive voiced frames required to confirm speech onset
286
287        - `max_speech_duration_s`:
288            - Type: float
289            - What: Hard cap on utterance duration in seconds; forces a return
290              even if the speaker never pauses
291
292        Optional:
293
294        - `inactivity_timeout`:
295            - Type: float | None
296            - What: Seconds to wait for speech onset before returning an empty buffer
297            - Default: None (wait indefinitely)
298
299        Returns:
300
301        - `buffer`:
302            - Type: list[int]
303            - What: Flat list of raw int16 PCM samples representing the captured utterance
304        """
305        self.ensure_vad()
306        frame_ms = self.frame_length / self.sample_rate * 1000
307        max_speech_frames = int(max_speech_duration_s * 1000 / frame_ms)
308        inactivity_frames = (
309            int(inactivity_timeout * 1000 / frame_ms)
310            if inactivity_timeout
311            else None
312        )
313
314        recorder = PvRecorder(
315            device_index=device_index, frame_length=self.frame_length
316        )
317
318        speech_buffer: list[list[int]] = []
319        pre_roll: list[list[int]] = []
320        in_speech: bool = False
321        voiced_frame_count: int = 0
322        silent_frame_count: int = 0
323        total_frames: int = 0
324
325        try:
326            recorder.start()
327            while stop_event is None or not stop_event.is_set():
328                frame = recorder.read()
329                total_frames += 1
330                tensor = (
331                    torch.tensor(frame, dtype=torch.float32).unsqueeze(0)
332                    / 32768.0
333                )
334
335                with torch.no_grad():
336                    speech_prob = self._vad_model(
337                        tensor, self.sample_rate
338                    ).item()
339
340                if not in_speech:
341                    if (
342                        inactivity_frames is not None
343                        and total_frames >= inactivity_frames
344                    ):
345                        break
346
347                    pre_roll.append(frame)
348                    if len(pre_roll) > speech_pad_frames:
349                        pre_roll.pop(0)
350
351                    if speech_prob >= speech_threshold:
352                        voiced_frame_count += 1
353                        if voiced_frame_count >= speech_pad_frames:
354                            in_speech = True
355                            silent_frame_count = 0
356                            speech_buffer = list(pre_roll)
357                    else:
358                        voiced_frame_count = 0
359                else:
360                    speech_buffer.append(frame)
361                    if speech_prob < silence_threshold:
362                        silent_frame_count += 1
363                    else:
364                        silent_frame_count = 0
365
366                    if (
367                        silent_frame_count >= silence_frames_threshold
368                        or len(speech_buffer) >= max_speech_frames
369                    ):
370                        break
371        finally:
372            recorder.stop()
373            recorder.delete()
374
375        return [sample for frame in speech_buffer for sample in frame]

379    def notify(
380        self,
381        message: str,
382        notification_type: str = "warning",
383        depth: int = 0,
384        force: bool = False,
385    ) -> None:
386        """
387        Usage:
388
389        - Creates a class based notification message
390
391        Requires:
392
393        - `message`:
394            - Type: str
395            - What: The message to warn users with
396            - Note: Messages with `{class_name}` and `{method_name}` in them are formatted appropriately
397
398        Optional:
399
400        - `notification_type`:
401            - Type: str
402            - What: The type of notification to send (warning, verbose or exception)
403            - Default: "warning"
404            - Note:
405                - "warning" prints a warning message
406                - "verbose" prints a verbose message only if `self.verbose=True`
407                - "exception" raises an exception with the message
408
409        - `depth`:
410            - Type: int
411            - What: The depth of the nth call below the top of the method stack
412            - Note: Depth starts at 0 (indicating the current method in the stack)
413            - Default: 0
414
415        - `force`:
416            - Type: bool
417            - What: If True, forces the message to print regardless of warning or verbose settings
418            - Default: False
419
420        Notes:
421
422        - If `self.warning_stack=True`, prints the stack trace alongside warning messages
423        - If `self.warnings=False`, suppresses all warning messages
424        - If `self.verbose=True`, enables verbose messages
425        """
426        notification_types = {
427            "warning": "WARNING",
428            "verbose": "",
429            "exception": "EXCEPTION",
430        }
431        message = f"{self.__class__.__name__}.{sys._getframe(depth).f_back.f_code.co_name} {notification_types.get(notification_type, '')}: {message}"
432        if notification_type == "exception":
433            raise Exception(message)
434        elif notification_type == "warning":
435            if self.__dict__.get("warnings", True) or force:
436                if self.__dict__.get("warning_stack", False):
437                    traceback.print_stack(limit=10)
438                print(message)
439        elif notification_type == "verbose" or force:
440            if self.__dict__.get("verbose", False):
441                print(message)
442        else:
443            raise Exception(
444                f"Invalid notification type. Must be one of: {list(notification_types.keys())}"
445            )

581class StreamSubprocess:
582    """
583    Usage:
584
585    - A context manager/iterator wrapper for subprocess.Popen that drains
586      stderr in the background and provides a thread-safe iterator over
587      stdout via a queue. This prevents deadlocks on Windows where a full
588      stderr pipe can block the process.
589
590    Example:
591
592    ```python
593    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, encoding='utf-8')
594    for line in StreamSubprocess(proc):
595        print(line)
596    ```
597    """
598
599    def __init__(self, proc: subprocess.Popen) -> None:
600        self.proc = proc
601        self.q = queue.Queue()
602        self._sentinel = object()
603        self.stderr_lines: list[str] = []
604
605        self._stdout_thread = threading.Thread(
606            target=self._reader, args=(self.proc.stdout,), daemon=True
607        )
608        self._stderr_thread = threading.Thread(
609            target=self._drain, args=(self.proc.stderr,), daemon=True
610        )
611
612        self._stdout_thread.start()
613        self._stderr_thread.start()
614
615    def _reader(self, pipe):
616        try:
617            if pipe:
618                for line in pipe:
619                    self.q.put(line)
620        except Exception:
621            pass
622        finally:
623            self.q.put(self._sentinel)
624
625    def _drain(self, pipe):
626        try:
627            if pipe:
628                for line in pipe:
629                    self.stderr_lines.append(line)
630        except Exception:
631            pass
632        finally:
633            if pipe:
634                pipe.close()
635
636    def __iter__(self):
637        return self
638
639    def __next__(self):
640        item = self.q.get()
641        if item is self._sentinel:
642            raise StopIteration
643        return item
644
645    def get(self, timeout: Optional[float] = None):
646        """
647        Get the next line from stdout with an optional timeout.
648        Returns None if the process has exited or the timeout is reached.
649        """
650        try:
651            item = self.q.get(timeout=timeout)
652            if item is self._sentinel:
653                return None
654            return item
655        except queue.Empty:
656            return None
657
658    def kill(self):
659        """Kills the underlying process and waits for threads to finish."""
660        try:
661            self.proc.kill()
662            self.proc.wait()
663        except Exception:
664            pass

667class StreamJsonCommand:
668    """
669    Usage:
670
671    - Launches a subprocess, writes optional stdin input, and provides a
672      thread-safe iterator over parsed JSON objects from stdout. Blank lines
673      and non-JSON lines are skipped silently. On Windows, wraps the command
674      in ``cmd.exe /c`` to ensure ``.cmd`` wrappers execute correctly.
675
676    Example:
677
678    ```python
679    for event in StreamJsonCommand(["gemini", "--output-format", "stream-json"], input_text="hello"):
680        print(event.get("type"))
681    ```
682    """
683
684    def __init__(
685        self, cmd: list[str], input_text: Optional[str] = None
686    ) -> None:
687        """
688        Usage:
689
690        - Launches the given command as a subprocess, writes optional input to
691          stdin, then immediately closes stdin so the process knows no more
692          input is coming.
693
694        Requires:
695
696        - `cmd`:
697            - Type: list[str]
698            - What: The command and arguments to execute.
699
700        Optional:
701
702        - `input_text`:
703            - Type: str | None
704            - What: Text to write to the process's stdin before closing it.
705            - Default: None
706        """
707        if os.name == "nt":
708            cmd = ["cmd.exe", "/c"] + cmd
709
710        proc = subprocess.Popen(
711            cmd,
712            stdin=subprocess.PIPE,
713            stdout=subprocess.PIPE,
714            stderr=subprocess.PIPE,
715            text=True,
716            encoding="utf-8",
717            errors="replace",
718            shell=False,
719        )
720
