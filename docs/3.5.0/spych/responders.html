  9    def __init__(
 10        self,
 11        spych_object: "Spych",
 12        listen_duration: int | float = 0,
 13        name: Optional[str] = None,
 14        spinner: Optional[CliSpinner] = None,
 15    ) -> None:
 16        """
 17        Usage:
 18
 19        - Base class for all responders. Handles the listen-transcribe-respond cycle,
 20          provides a consistent interface for subclasses to implement, and includes
 21          a rich terminal UI and animated spinner.
 22
 23        - Subclasses only need to implement `respond(user_input: str) -> str`.
 24          All CLI chrome (spinner, dividers, timing, response box) is handled here.
 25
 26        - Public helper methods are available inside `respond()` for common
 27          UI needs without importing CLI internals:
 28
 29            - `self.spinner.start(message)`          — restart spinner after a pause
 30            - `self.spinner.stop()`                  — stop spinner (e.g. before printing)
 31            - `self.print_info(message, color)`      — print a styled info line
 32
 33        Requires:
 34
 35        - `spych_object`:
 36            - Type: Spych
 37            - What: An initialized Spych instance used to record and transcribe audio
 38              after the wake word is detected
 39
 40        Optional:
 41
 42        - `listen_duration`:
 43            - Type: int | float
 44            - What: The number of seconds to listen for after the wake word is detected
 45            - Default: 0 (listen indefinitely until silence is detected)
 46
 47        - `name`:
 48            - Type: str
 49            - What: A custom name for the responder to use in printed messages
 50            - Default: The class name of the responder (e.g., "Ollama")
 51
 52        - `spinner`:
 53            - Type: CliSpinner | None
 54            - What: An externally-owned spinner to share across multiple responders.
 55              When provided (e.g. by SpychOrchestrator), all responders drive the same
 56              spinner so their output never interleaves. When None, a private spinner is
 57              created, preserving the original single-responder behaviour.
 58            - Default: None
 59
 60        Notes:
 61
 62        - Subclasses must implement the `respond` method.
 63        - The `__call__` method orchestrates the full voice listen -> transcribe -> respond
 64          cycle; use `text_input` for the typed equivalent.
 65        """
 66        self.spych_object = spych_object
 67        self.listen_duration = listen_duration
 68        self.name = name if name else self.__class__.__name__
 69        # Accept an injected shared spinner or create a private one.
 70        self.spinner: CliSpinner = (
 71            spinner if spinner is not None else CliSpinner()
 72        )
 73        self._start_time: float = 0.0
 74

 80        """
 81        Usage:
 82
 83        - Print a divider and reset the spinner after each complete cycle to indicate
 84          that the responder is waiting for the next wake word. Called automatically at
 85          the end of `__call__` and `text_input`, but can also be called manually.
 86
 87        Optional:
 88
 89        - `divider`:
 90            - Type: bool
 91            - What: Whether to print a divider line before restarting the spinner
 92            - Default: True
 93        """
 94        if divider:
 95            CliPrinter.divider()
 96        self.spinner.start("Waiting for wake word", spinner=Spinner.BRAILLE)

131            - What: Elapsed time in seconds since the tool started (if available)
132            - Default: None

323
324        Requires:

327            - Type: str
328            - What: The response string returned by the respond method

365        - This method is called automatically at the end of each listen cycle.

371    def __call__(self) -> str:
372        """
373        Usage:
374
375        - Executes one full wake-triggered cycle: listens, transcribes, responds,
376          and prints.
377
378        Returns:
379
380        - `response`:
381            - Type: str
382            - What: The response string returned by `respond`
383        """
384        self.on_listen_start()
385        user_input = self.spych_object.listen(duration=self.listen_duration)
386        self.on_listen_end()
387        if not user_input:
388            self.wait_for_next_wake_word(divider=False)
389            return ""
390        self.on_user_input(user_input)
391        try:
392            self.on_before_respond(user_input)
393            response = self.respond(user_input)
394            self.on_after_respond(user_input, response)
395        except Exception as exc:
396            self.spinner.stop()
397            print(f"  {theme.error}✗  Error: {exc}{theme.reset}\n")
398            return ""
399        self.on_response(response)
400        self.wait_for_next_wake_word()
401        return response

403    def ready_message(self, show_wait_for_wake: bool = True, **kwargs) -> None:
404        """
405        Formats and prints the ready message when the responder is initialized,
406        showing the wake words and terminate words.
407
408        Optional:
409
410        - `show_wait_for_wake`:
411            - Type: bool
412            - What: Whether to include "Waiting for wake word" in the ready message
413            - Default: True
414
415        - `**kwargs`:
416            - Type: dict
417            - What: Additional keyword arguments to display in the ready message
418
419        """
420        CliPrinter.header(self.name)
421        CliPrinter.kwarg_inputs(**kwargs)
422        CliPrinter.empty_line()
423        if show_wait_for_wake:
424            self.wait_for_next_wake_word()
