 10def resolve_cmd(name: str) -> str:
 11    """
 12    Usage:
 13
 14    - Resolves a CLI command name to its full executable path using shutil.which.
 15      On Windows, Node.js CLI tools are installed as .cmd wrappers (e.g. gemini.cmd).
 16      While shutil.which returns the full path including the extension,
 17      subprocess.Popen on Windows often requires shell=True to correctly
 18      execute these batch wrappers.
 19
 20    Requires:
 21
 22    - `name`:
 23        - Type: str
 24        - What: The CLI command name to resolve (e.g. "gemini", "claude")
 25
 26    Returns:
 27
 28    - `path`:
 29        - Type: str
 30        - What: The full path to the executable if found, otherwise the original name
 31    """
 32    return shutil.which(name) or name

 35def supports_unicode() -> bool:
 36    """
 37    Check if the terminal likely supports Unicode icons (like braille or emoji).
 38    On Windows, we check for 'utf-8' encoding or typical modern terminal indicators.
 39    """
 40    if sys.platform != "win32":
 41        return True
 42    # Modern Windows Terminal or VS Code terminal usually support Unicode
 43    if os.environ.get("WT_SESSION") or os.environ.get("VSCODE_PID"):
 44        return True
 45    # If the output encoding is explicitly set to utf-8, we trust it
 46    if getattr(sys.stdout, "encoding", "").lower() == "utf-8":
 47        return True
 48    return False

 51def get_cache_dir(folder="voices") -> str:
 52    """Returns the path to the project's voice cache directory."""
 53    if os.name == "nt":  # Windows
 54        base_dir = os.environ.get("LOCALAPPDATA", os.path.expanduser("~"))
 55    elif sys.platform == "darwin":  # macOS
 56        base_dir = os.path.expanduser("~/Library/Caches")
 57    else:  # Linux/Unix
 58        base_dir = os.environ.get(
 59            "XDG_CACHE_HOME", os.path.expanduser("~/.cache")
 60        )
 61
 62    path = os.path.join(base_dir, "spych", folder)
 63    os.makedirs(path, exist_ok=True)
 64    return path

 67def get_setting(key: str, default: Any = None) -> Any:
 68    """Returns a setting from the cache."""
 69    path = os.path.join(get_cache_dir("settings"), "settings.json")
 70    if not os.path.exists(path):
 71        return default
 72    with open(path, "r", encoding="utf-8") as f:
 73        try:
 74            settings = json.load(f)
 75        except json.JSONDecodeError:
 76            return default
 77    return settings.get(key, default)

164def resolve_whisper_device(device: str) -> str:
165    """
166    Usage:
167
168    - Resolves the whisper device string, selecting "cuda" or "cpu" automatically
169      when "auto" is passed.
170    - "cuda" is chosen when torch reports a CUDA device is available; if the
171      CUDA libraries cannot actually be loaded at runtime, `load_whisper_model`
172      will catch the resulting `RuntimeError` and transparently fall back to CPU.
173
174    Requires:
175
176    - `device`:
177        - Type: str
178        - What: Requested device — "auto", "cpu", or "cuda"
179
180    Returns:
181
182    - `resolved`:
183        - Type: str
184        - What: "cpu" or "cuda"
185    """
186    if device != "auto":
187        return device
188    if torch.cuda.is_available():
189        return "cuda"
190    return "cpu"

193def load_whisper_model(
194    whisper_model: str,
195    device: str,
196    compute_type: str,
197) -> "WhisperModel":
198    """
199    Usage:
200
201    - Constructs a `faster_whisper.WhisperModel` and transparently falls back
202      to CPU when CUDA is requested but unavailable at runtime (e.g. when
203      `torch.cuda.is_available()` returns True but the CUDA libraries such as
204      `libcublas.so.12` are missing or incompatible).
205    - When `device` is \"cuda\", a silent dummy transcription is run immediately
206      after construction to force eager CUDA initialisation. ctranslate2
207      initialises CUDA lazily, so the constructor itself never raises even when
208      the libraries are broken — the error only surfaces at the first encode
209      call. The probe catches that error here and falls back to CPU before the
210      model is handed to any caller.
211
212    Requires:
213
214    - `whisper_model`:
215        - Type: str
216        - What: The faster-whisper model name (e.g. \"base.en\", \"small\")
217
218    - `device`:
219        - Type: str
220        - What: The device to attempt first — \"cpu\" or \"cuda\"
221
222    - `compute_type`:
223        - Type: str
224        - What: The compute type (e.g. \"int8\", \"float16\", \"float32\")
225
226    Returns:
227
228    - `model`:
229        - Type: WhisperModel
230        - What: A loaded faster-whisper model instance, guaranteed to run on
231          either the requested device or CPU if the requested device failed.
232
233    Notes:
234
235    - The dummy probe is a zero-valued 0.1 s float32 buffer, which is the
236      smallest input ctranslate2 will encode without complaining. Its output
237      is discarded; only the side-effect of triggering CUDA initialisation matters.
238    - If the CPU fallback also raises, the exception propagates normally.
239    """
240    from faster_whisper import WhisperModel
241
242    def _build(dev: str, ct: str) -> WhisperModel:
243        return WhisperModel(whisper_model, device=dev, compute_type=ct)
244
245    def _probe(model: WhisperModel) -> None:
246        dummy = np.zeros(int(16000 * 0.1), dtype=np.float32)
247        list(model.transcribe(dummy, beam_size=1)[0])
248
249    try:
250        model = _build(device, compute_type)
251        if device != "cpu":
252            _probe(model)
253        return model
254    except RuntimeError as e:
255        if device != "cpu":
256            print(
257                f"WARNING: Could not use WhisperModel on '{device}' "
258                f"({e}). Falling back to CPU."
259            )
260            return _build("cpu", "int8")
261        raise

264class Recorder:
265    def __init__(
266        self,
267        sample_rate: int = 16000,
268        frame_length: int = 512,
269    ) -> None:
270        """
271        Usage:
272
273        - Initializes a recorder that wraps PvRecorder for fixed-duration recording
274          and Silero VAD for utterance-gated recording. The VAD model is loaded once
275          at first call and reused across all `record_vad` calls.
276
277        Optional:
278
279        - `sample_rate`:
280            - Type: int
281            - What: The sample rate in Hz used for both recording and VAD inference
282            - Default: 16000
283            - Note: Should match the sample rate expected by your downstream model
284
285        - `frame_length`:
286            - Type: int
287            - What: The number of samples per frame read from PvRecorder
288            - Default: 512
289            - Note: Also used as Silero VAD's native 16kHz window (~32ms per frame)
290        """
291        self.sample_rate = sample_rate
292        self.frame_length = frame_length
293        self.frame_ms = self.frame_length / self.sample_rate * 1000
294
295    def ensure_vad(self) -> None:
296        if not hasattr(self, "_vad_model"):
297            self._vad_model = load_silero_vad()
298            self._vad_model.eval()
299
300    def record(
301        self,
302        device_index: int,
303        duration: Union[int, float],
304    ) -> list[int]:
305        """
306        Usage:
307
308        - Records audio from a microphone for a fixed duration and returns a raw
309          PCM buffer.
310
311        Requires:
312
313        - `device_index`:
314            - Type: int
315            - What: The index of the microphone device to record from
316            - Note: Use `-1` to select the system default input device
317
318        - `duration`:
319            - Type: int | float
320            - What: The number of seconds to record
321
322        Returns:
323
324        - `buffer`:
325            - Type: list[int]
326            - What: A flat list of raw int16 PCM samples at the configured sample rate
327        """
328        recorder = PvRecorder(
329            device_index=device_index, frame_length=self.frame_length
330        )
331        try:
332            recorder.start()
333            frames = int(self.sample_rate * duration / self.frame_length)
334            buffer = []
335            for _ in range(frames):
336                buffer.extend(recorder.read())
337        except Exception as e:
338            raise e
339        finally:
340            recorder.stop()
341            recorder.delete()
342        return buffer

344    def record_vad(
345        self,
346        device_index: int,
347        speech_threshold: float,
348        silence_threshold: float,
349        silence_frames_threshold: int,
350        speech_pad_frames: int,
351        max_speech_duration_s: float,
352        inactivity_timeout: Optional[float] = None,
353        stop_event=None,
354    ) -> list[int]:
355        """
356        Usage:
357
358        - Records from the microphone using the cached Silero VAD model to detect
359          a single complete utterance and returns the raw PCM buffer. Blocks until
360          speech is detected and then silence confirms the utterance is complete.
361
362        Requires:
363
364        - `device_index`:
365            - Type: int
366            - What: PvRecorder device index; -1 uses system default
367
368        - `speech_threshold`:
369            - Type: float
370            - What: Silero probability above which a frame is considered speech onset
371
372        - `silence_threshold`:
373            - Type: float
374            - What: Silero probability below which a frame is considered silence
375              during an active speech segment; must be less than `speech_threshold`
376              to create a hysteresis band
377
378        - `silence_frames_threshold`:
379            - Type: int
380            - What: Consecutive below-threshold frames required to confirm end of utterance
381
382        - `speech_pad_frames`:
383            - Type: int
384            - What: Pre-roll frames captured before onset confirmation; also the
385              number of consecutive voiced frames required to confirm speech onset
386
387        - `max_speech_duration_s`:
388            - Type: float
389            - What: Hard cap on utterance duration in seconds; forces a return
390              even if the speaker never pauses
391
392        Optional:
393
394        - `inactivity_timeout`:
395            - Type: float | None
396            - What: Seconds to wait for speech onset before returning an empty buffer
397            - Default: None (wait indefinitely)
398
399        Returns:
400
401        - `buffer`:
402            - Type: list[int]
403            - What: Flat list of raw int16 PCM samples representing the captured utterance
404        """
405        self.ensure_vad()
406        frame_ms = self.frame_length / self.sample_rate * 1000
407        max_speech_frames = int(max_speech_duration_s * 1000 / frame_ms)
408        inactivity_frames = (
409            int(inactivity_timeout * 1000 / frame_ms)
410            if inactivity_timeout
411            else None
412        )
413
414        recorder = PvRecorder(
415            device_index=device_index, frame_length=self.frame_length
416        )
417
418        speech_buffer: list[list[int]] = []
419        pre_roll: list[list[int]] = []
420        in_speech: bool = False
421        voiced_frame_count: int = 0
422        silent_frame_count: int = 0
423        total_frames: int = 0
424
425        try:
426            recorder.start()
427            while stop_event is None or not stop_event.is_set():
428                frame = recorder.read()
429                total_frames += 1
430                tensor = (
431                    torch.tensor(frame, dtype=torch.float32).unsqueeze(0)
432                    / 32768.0
433                )
434
435                with torch.no_grad():
436                    speech_prob = self._vad_model(
437                        tensor, self.sample_rate
438                    ).item()
439
440                if not in_speech:
441                    if (
442                        inactivity_frames is not None
443                        and total_frames >= inactivity_frames
444                    ):
445                        break
446
447                    pre_roll.append(frame)
448                    if len(pre_roll) > speech_pad_frames:
449                        pre_roll.pop(0)
450
451                    if speech_prob >= speech_threshold:
452                        voiced_frame_count += 1
453                        if voiced_frame_count >= speech_pad_frames:
454                            in_speech = True
455                            silent_frame_count = 0
456                            speech_buffer = list(pre_roll)
457                    else:
458                        voiced_frame_count = 0
459                else:
460                    speech_buffer.append(frame)
461                    if speech_prob < silence_threshold:
462                        silent_frame_count += 1
463                    else:
464                        silent_frame_count = 0
465
466                    if (
467                        silent_frame_count >= silence_frames_threshold
468                        or len(speech_buffer) >= max_speech_frames
469                    ):
470                        break
471        finally:
472            recorder.stop()
473            recorder.delete()
474
475        return [sample for frame in speech_buffer for sample in frame]

478class Notify:
479    def notify(
480        self,
481        message: str,
482        notification_type: str = "warning",
483        depth: int = 0,
484        force: bool = False,
485    ) -> None:
486        """
487        Usage:
488
489        - Creates a class based notification message
490
491        Requires:
492
493        - `message`:
494            - Type: str
495            - What: The message to warn users with
496            - Note: Messages with `{class_name}` and `{method_name}` in them are formatted appropriately
497
498        Optional:
499
500        - `notification_type`:
501            - Type: str
502            - What: The type of notification to send (warning, verbose or exception)
503            - Default: "warning"
504            - Note:
505                - "warning" prints a warning message
506                - "verbose" prints a verbose message only if `self.verbose=True`
507                - "exception" raises an exception with the message
508
509        - `depth`:
510            - Type: int
511            - What: The depth of the nth call below the top of the method stack
512            - Note: Depth starts at 0 (indicating the current method in the stack)
513            - Default: 0
514
515        - `force`:
516            - Type: bool
517            - What: If True, forces the message to print regardless of warning or verbose settings
518            - Default: False
519
520        Notes:
521
522        - If `self.warning_stack=True`, prints the stack trace alongside warning messages
523        - If `self.warnings=False`, suppresses all warning messages
524        - If `self.verbose=True`, enables verbose messages
525        """
526        notification_types = {
527            "warning": "WARNING",
528            "verbose": "",
529            "exception": "EXCEPTION",
530        }
531        if notification_type not in notification_types:
532            raise Exception(
533                f"Invalid notification type. Must be one of: {list(notification_types.keys())}"
534            )
535        message = f"{self.__class__.__name__}.{sys._getframe(depth).f_back.f_code.co_name} {notification_types.get(notification_type, '')}: {message}"
536        if notification_type == "exception":
537            raise Exception(message)
538        elif notification_type == "warning":
539            if self.__dict__.get("warnings", True) or force:
540                if self.__dict__.get("warning_stack", False):
541                    traceback.print_stack(limit=10)
542                print(message)
543        elif notification_type == "verbose":
544            if self.__dict__.get("verbose", False) or force:
545                print(message)

681class StreamSubprocess:
682    """
683    Usage:
684
685    - A context manager/iterator wrapper for subprocess.Popen that drains
686      stderr in the background and provides a thread-safe iterator over
687      stdout via a queue. This prevents deadlocks on Windows where a full
688      stderr pipe can block the process.
689
690    Example:
691
692    ```python
693    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, encoding='utf-8')
694    for line in StreamSubprocess(proc):
695        print(line)
696    ```
