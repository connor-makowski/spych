 21    def __init__(
 22        self,
 23        entries: list[OrchestratorEntry],
 24        spych_wake_kwargs: Optional[dict] = None,
 25    ) -> None:
 26        """
 27        Usage:
 28
 29        - Runs one or more responders under a single SpychWake and a single shared
 30          spinner. Each entry binds a responder to its own wake words.
 31
 32        Requires:
 33
 34        - `entries`:
 35            - Type: list[OrchestratorEntry]
 36            - What: Each dict registers one responder. Required keys:
 37                - "responder"      : BaseResponder instance
 38                - "wake_words"     : list[str] — words that trigger this responder
 39              Optional keys:
 40                - "terminate_words": list[str] — words that stop the entire orchestrator
 41                                     Default: ["terminate"]
 42            - Note: Wake words must be unique across all entries.
 43              Terminate words are merged — any terminate word from any entry stops
 44              the whole orchestrator.
 45            - Example:
 46                [
 47                    {
 48                        "responder": claude_responder,
 49                        "wake_words": ["claude", "cloud"],
 50                        "terminate_words": ["terminate"],
 51                    },
 52                    {
 53                        "responder": ollama_responder,
 54                        "wake_words": ["llama", "ollama"],
 55                    },
 56                ]
 57
 58        Optional:
 59
 60        - `spych_wake_kwargs`:
 61            - Type: dict | None
 62            - What: Extra kwargs forwarded to SpychWake (e.g. whisper_model,
 63              wake_listener_count). Keys present here override the defaults.
 64            - Default: None
 65        """
 66        if not entries:
 67            raise ValueError("SpychOrchestrator requires at least one entry.")
 68
 69        self.entries = self.build_entries(entries)
 70        # Inherit the dashboard from the first responder if one was injected.
 71        self._dashboard: Optional[AgentDashboard] = (
 72            getattr(self.entries[0]["responder"], "dashboard", None)
 73            if self.entries
 74            else None
 75        )
 76        self.spinner = self.build_spinner()
 77        self.wake_word_map, self.terminate_words = self.build_wake_word_map()
 78        self.spych_wake = self.build_spych_wake(spych_wake_kwargs)

 84    def build_entries(
 85        self, raw_entries: list[OrchestratorEntry]
 86    ) -> list[OrchestratorEntry]:
 87        """
 88        Usage:
 89
 90        - Validates and normalises the raw entries list passed to __init__,
 91          ensuring required keys are present and filling in defaults.
 92
 93        Requires:
 94
 95        - `raw_entries`:
 96            - Type: list[OrchestratorEntry]
 97            - What: The unvalidated entries passed by the caller
 98
 99        Returns:
100
101        - `entries`:
102            - Type: list[OrchestratorEntry]
103            - What: Validated and normalised entries with defaults applied
104        """
105        entries: list[OrchestratorEntry] = []
106        for raw in raw_entries:
107            if "responder" not in raw:
108                raise ValueError("Each entry must contain a 'responder' key.")
109            if "wake_words" not in raw or not raw["wake_words"]:
110                raise ValueError(
111                    "Each entry must contain a non-empty 'wake_words' key."
112                )
113            entries.append(
114                {
115                    "responder": raw["responder"],
116                    "wake_words": list(raw["wake_words"]),
117                    "terminate_words": list(
118                        raw.get("terminate_words", ["terminate"])
119                    ),
120                }
121            )
122        return entries

124    def build_spinner(self) -> CliSpinner:
125        """
126        Usage:
127
128        - Creates a single shared CliSpinner and injects it into every registered
129          responder so all UI output goes through one instance and lines never
130          interleave.
131
132        Returns:
133

145    def build_wake_word_map(
146        self,
147    ) -> tuple[dict[str, "BaseResponder"], list[str]]:
148        """
149        Usage:
150
151        - Iterates over all entries to build a flat wake-word-to-responder map
152          and a deduplicated list of terminate words. Each responder's __call__
153          is wrapped so the most recently triggered responder is always tracked.
154
155        - Raises if any wake word is registered by more than one responder.
156
157        Returns:
158
159        - `(wake_word_map, terminate_words)`:
160            - Type: tuple[dict[str, BaseResponder], list[str]]
161            - What: The resolved wake word map and merged terminate word list
162        """
163        terminate_words: list[str] = []
164        wake_word_map: dict[str, "BaseResponder"] = {}
165
166        for entry in self.entries:
167            for word in entry["wake_words"]:
168                w = word.lower()
169                if w in wake_word_map:
170                    raise ValueError(
171                        f"Wake word '{w}' is registered by more than one responder."
172                    )
173                wake_word_map[w] = entry["responder"]
174            for word in entry["terminate_words"]:
175                w = word.lower()
176                if w not in terminate_words:
177                    terminate_words.append(w)
178
179        def make_tracked_call(
180            resp: "BaseResponder", entry_wake_words: list[str]
181        ):
182            original = resp.__call__
183
184            def tracked():
185                self.last_responder = resp
186                if self._dashboard is not None:
187                    self._dashboard.set_agent(
188                        name=resp.name,
189                        wake_words=entry_wake_words,
190                        responder_name=getattr(resp, "display_name", None)
191                        or resp.__class__.__name__,
192                        response_style=getattr(resp, "response_style", ""),
193                        use_speaker=getattr(resp, "use_speaker", False),
194                        speaker_voice=getattr(resp, "speaker_voice", ""),
195                        user_name=getattr(resp, "user_name", "User"),
196                    )
197                original()
198
199            return tracked
200
201        # Build a per-entry wake-word map to preserve each entry's wake_words for set_agent.
202        entry_wake_words_by_resp: dict[str, list[str]] = {}
203        for entry in self.entries:
204            for word in entry["wake_words"]:
205                entry_wake_words_by_resp[word.lower()] = entry["wake_words"]
206
207        tracked_wake_map: dict[str, callable] = {
208            word: make_tracked_call(
209                resp, entry_wake_words_by_resp.get(word, [])
210            )
211            for word, resp in wake_word_map.items()
212        }
213
214        self.last_responder: "BaseResponder" = self.entries[0]["responder"]
215        return tracked_wake_map, terminate_words

217    def find_shareable_wake_model(self, kwargs: dict) -> Optional[object]:
218        """
219        Usage:
220
221        - Checks whether the first entry's responder already has a loaded
222          command-transcription WhisperModel (via its `spych_object`) that is
223          configured identically to what SpychWake would otherwise construct
224          for wake-word spotting. If so, that model can be reused instead of
225          loading a second, redundant copy into memory.
226
227        Requires:
228
229        - `kwargs`:
230            - Type: dict
231            - What: The (possibly empty) kwargs about to be forwarded to
232              SpychWake, used to determine the effective whisper_model/
233              whisper_device/whisper_compute_type it would otherwise use.
234
235        Returns:
236
237        - `wake_model`:
238            - Type: faster_whisper.WhisperModel | None
239            - What: The shareable model instance, or None if no match was found
240              (e.g. different model name/device/compute_type, or the caller
241              already passed an explicit `wake_model_instance`).
242
243        Notes:
244
245        - Only the first entry is considered, mirroring how the dashboard is
246          also inherited from the first responder — in the common case (a
247          single agent, or `spych multi` where every responder shares one
248          Spych instance) that's representative of the whole orchestrator.
249        """
250        if "wake_model_instance" in kwargs:
251            return None
252
253        spych_object = getattr(
254            self.entries[0]["responder"], "spych_object", None
255        )
256        if spych_object is None:
257            return None
258
259        defaults = {
260            name: param.default
261            for name, param in inspect.signature(
262                SpychWake.__init__
263            ).parameters.items()
264        }
265        whisper_model = kwargs.get("whisper_model", defaults["whisper_model"])
266        whisper_device = resolve_whisper_device(
267            kwargs.get("whisper_device", defaults["whisper_device"])
268        )
269        whisper_compute_type = kwargs.get(
270            "whisper_compute_type", defaults["whisper_compute_type"]
271        )
272
273        if (
274            spych_object.whisper_model == whisper_model
275            and spych_object.whisper_device == whisper_device
276            and spych_object.whisper_compute_type == whisper_compute_type
277        ):
278            return spych_object.wake_model
279        return None

281    def build_spych_wake(self, spych_wake_kwargs: Optional[dict]) -> SpychWake:
282        """
283        Usage:
284
285        - Constructs the SpychWake instance using the merged wake word map and
286          terminate word list built by build_wake_word_map().
287        - Reuses the first responder's command-transcription WhisperModel for
288          wake-word spotting when its model/device/compute_type already match
289          what SpychWake would otherwise load, avoiding a redundant second
290          model in memory.
291
292        Requires:
293
294        - `spych_wake_kwargs`:
295            - Type: dict | None
296            - What: Extra kwargs forwarded to SpychWake. whisper_model uses
297              SpychWake's own fast "small.en" default if not provided.
298
299        Returns:
300
301        - `spych_wake`:
302            - Type: SpychWake
303            - What: The configured SpychWake instance ready to be started
304        """
305        kwargs = spych_wake_kwargs or {}
306        shared_model = self.find_shareable_wake_model(kwargs)
307        if shared_model is not None:
308            kwargs = {**kwargs, "wake_model_instance": shared_model}
309        return SpychWake(
310            wake_word_map=self.wake_word_map,
311            terminate_words=self.terminate_words,
312            on_terminate=self.on_terminate,
313            **kwargs,

320    def on_terminate(self) -> None:
321        """
322        Usage:
323
324        - Called by SpychWake when any terminate word is detected. Calls
325          on_terminate() on every registered responder.
326        """
327        for entry in self.entries:
328            try:
329                entry["responder"].on_terminate()
330            except Exception:
331                pass

337    def start(self) -> None:
338        """
339        Usage:
340
341        - Prints the ready header for every registered responder, then starts
342          the SpychWake voice listener. Blocks until a terminate word is spoken
343          or ctrl+c is pressed.
344
345        Notes:
346
347        - If a dashboard is active, it is always stopped (restoring the
348          terminal) before this method returns or raises, regardless of how
349          `spych_wake.start()` exits.
350        """
351        for entry in self.entries:
352            if not entry["responder"].healthcheck():
353                CliPrinter.info(
354                    f"{entry['responder'].name} healthcheck failed.",
355                    color=theme.error,
356                )
357                return
358            entry["responder"].ready_message(
359                show_wait_for_wake=False,
360                wake_words=entry["wake_words"],
361                terminate_words=entry["terminate_words"],
362                responder=entry["responder"].__class__.__name__,
363            )
364
365        if self._dashboard is not None:
366            self._dashboard.start()
367        else:
368            CliPrinter.divider("─", 60, theme.accent)
369        self.spinner.start("Waiting for wake word", spinner=Spinner.BRAILLE)
370
371        try:
372            self.spych_wake.start()
373        except KeyboardInterrupt:
374            self.on_terminate()
375        finally:
376            if self._dashboard is not None:
377                self._dashboard.stop()

def build_wake_word_map(self) -> "tuple[dict[str, 'BaseResponder'], list[str]]": 
146    def build_wake_word_map(
147        self,
148    ) -> tuple[dict[str, "BaseResponder"], list[str]]:
149        """
150        Usage:
151
152        - Iterates over all entries to build a flat wake-word-to-responder map
153          and a deduplicated list of terminate words. Each responder's __call__
154          is wrapped so the most recently triggered responder is always tracked.
155
156        - Raises if any wake word is registered by more than one responder.
157
158        Returns:
159
160        - `(wake_word_map, terminate_words)`:
161            - Type: tuple[dict[str, BaseResponder], list[str]]
162            - What: The resolved wake word map and merged terminate word list
163        """
164        terminate_words: list[str] = []
165        wake_word_map: dict[str, "BaseResponder"] = {}
166
167        for entry in self.entries:
168            for word in entry["wake_words"]:
169                w = word.lower()
170                if w in wake_word_map:
171                    raise ValueError(
172                        f"Wake word '{w}' is registered by more than one responder."
173                    )
174                wake_word_map[w] = entry["responder"]
175            for word in entry["terminate_words"]:
176                w = word.lower()
177                if w not in terminate_words:
178                    terminate_words.append(w)
179
180        def make_tracked_call(
181            resp: "BaseResponder", entry_wake_words: list[str]
182        ):
183            original = resp.__call__
184
185            def tracked():
186                self.last_responder = resp
187                if self._dashboard is not None:
188                    self._dashboard.set_agent(
189                        name=resp.name,
190                        wake_words=entry_wake_words,
191                        responder_name=getattr(resp, "display_name", None)
192                        or resp.__class__.__name__,
193                        response_style=getattr(resp, "response_style", ""),
194                        use_speaker=getattr(resp, "use_speaker", False),
195                        speaker_voice=getattr(resp, "speaker_voice", ""),
196                        user_name=getattr(resp, "user_name", "User"),
197                    )
198                original()
199
200            return tracked
201
202        # Build a per-entry wake-word map to preserve each entry's wake_words for set_agent.
203        entry_wake_words_by_resp: dict[str, list[str]] = {}
204        for entry in self.entries:
205            for word in entry["wake_words"]:
206                entry_wake_words_by_resp[word.lower()] = entry["wake_words"]
207
208        tracked_wake_map: dict[str, callable] = {
209            word: make_tracked_call(
210                resp, entry_wake_words_by_resp.get(word, [])
211            )
212            for word, resp in wake_word_map.items()
213        }
214
215        self.last_responder: "BaseResponder" = self.entries[0]["responder"]
216        return tracked_wake_map, terminate_words
