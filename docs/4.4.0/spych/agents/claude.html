 17def _extract_tool_detail(tool_name: str, tool_input: dict) -> str:
 18    """
 19    Usage:
 20
 21    - Returns a short human-readable summary of the most relevant argument for
 22      a Claude Code tool call. Used to populate the ``detail`` field of tool
 23      events so the terminal displays e.g. ``Read  /path/to/file.py`` instead
 24      of a raw JSON blob.
 25
 26    Requires:
 27
 28    - `tool_name`:
 29        - Type: str
 30        - What: The name of the tool being invoked (e.g. "Read", "Bash").
 31
 32    - `tool_input`:
 33        - Type: dict
 34        - What: The parsed input parameters for the tool call.
 35
 36    Returns:
 37
 38    - `detail`:
 39        - Type: str
 40        - What: A concise, display-safe string describing the key argument.
 41          Empty string when no meaningful detail can be extracted.
 42    """
 43    _MAP: dict[str, Any] = {
 44        "Read": lambda i: i.get("file_path", ""),
 45        "Edit": lambda i: i.get("file_path", ""),
 46        "Write": lambda i: i.get("file_path", ""),
 47        "Bash": lambda i: (i.get("command", "") or "")[:80],
 48        "Grep": lambda i: (
 49            f'"{i.get("pattern", "")}"'
 50            + (f' in {i.get("path", "")}' if i.get("path") else "")
 51        ),
 52        "Glob": lambda i: i.get("pattern", ""),
 53        "WebFetch": lambda i: i.get("url", ""),
 54        "WebSearch": lambda i: i.get("query", ""),
 55        "Agent": lambda i: i.get("description", "subagent"),
 56        "NotebookEdit": lambda i: i.get(
 57            "notebook_path", i.get("cell_type", "")
 58        ),
 59        "TodoWrite": lambda i: "updating todos",
 60    }
 61    fn = _MAP.get(tool_name)
 62    if fn:
 63        return fn(tool_input) or ""
 64    # Generic fallback: first non-trivial string value in the input dict
 65    for v in tool_input.values():
 66        if isinstance(v, str) and v:
 67            return v[:80]
 68    return ""

169    def respond(
170        self, user_input: str, is_continuation: bool = False
171    ) -> AgentResponse:
172        """
173        Spawns _sdk_worker.py as a subprocess, writes the request payload to
174        its stdin, then reads newline-delimited JSON events from its stdout.
175
176        Tool start/end events are fired live as they arrive so the user sees
177        real-time feedback. The final result is parsed into an AgentResponse.
178        """
179        is_first = self._first_call
180        self._first_call = False
181
182        prompt = user_input
183        if is_continuation:
184            prompt = "Please continue."
185
186        payload = json.dumps(
187            {
188                "user_input": self.format_prompt(prompt),
189                "is_first": is_first,
190                "continue_conversation": self.continue_conversation,
191                "last_session_id": self._last_session_id,
192                "setting_sources": self.setting_sources,
193            }
194        )
195
196        # print(payload)
197
198        stream = StreamJsonCommand(
199            [sys.executable, str(CLAUDE_SDK_WORKER_PATH)], input_text=payload
200        )
201
202        final_result = ""
203        active_tools: dict[str, tuple[str, float]] = {}
204
205        try:
206            for event in stream:
207                etype = event.get("type")
208
209                if etype == "session":
210                    self._last_session_id = event.get("id")
211
212                elif etype == "system":
213                    # print(raw_line)
214                    pass
215
216                elif etype == "tool_start":
217                    tool_id = event.get("id")
218                    tool_name = event.get("name")
219                    if tool_id is None or tool_name is None:
220                        continue
221                    raw_input = event.get("input", {})
222                    # print(f"Tool '{tool_name}' started with input: {json.dumps(raw_input)}")
223                    active_tools[tool_id] = (tool_name, time.time())
224                    if self.show_tool_events:
225                        detail = _extract_tool_detail(tool_name, raw_input)
226                        self.tool_event(
227                            tool_name,
228                            "running",
229                            is_running=True,
230                            detail=detail or None,
231                        )
232
233                elif etype == "tool_end":
234                    tool_id = event.get("id")
235                    if tool_id in active_tools:
236                        tool_name, start = active_tools.pop(tool_id)
237                        elapsed = time.time() - start
238                        if self.show_tool_events:
239                            self.tool_event(
240                                tool_name,
241                                "done",
242                                is_running=False,
243                                elapsed=elapsed,
244                            )
245
246                elif etype == "result":
247                    final_result = event.get("text", "")
248
249                elif etype == "error":
250                    final_result = (
251                        f"Error: {event.get('text', 'unknown error')}"
252                    )
253        finally:
254            stream.kill()
255
256        if not final_result:
257            stderr_text = "".join(stream.stderr_lines).strip()
258            if stderr_text:
259                final_result = f"Error: {stderr_text}"
260
261        return self.parse_output(final_result)

461          never printed directly, to avoid double-printing.
462        - Claude Code CLI must be installed: npm install -g @anthropic-ai/claude-code

465        super().__init__(

490    def __run_turn__(
491        self,
492        user_input: str,
493        is_first: bool,
494        active_tools: dict[str, float],
495        print_assistant: bool,
496    ) -> tuple[bool, str]:
497        """
498        Run one `claude -p` subprocess turn.
499
500        Args:
501            user_input:      Prompt to send to the CLI.
502            is_first:        True only on the very first ever call (skips --resume).
503            active_tools:    Shared dict of tool_name -> start_time across turns.
504            print_assistant: If True, print intermediate assistant text live.
505                             Set False on the final turn to avoid double-printing
506                             (the base class prints the return value of respond()).
507
508        Returns:
509            (needs_continuation, result_text)
510            - needs_continuation=True means the result contained a raw
511              <tool_call> block and should be re-submitted on the same session.
512            - result_text is either the raw tool-call text (to re-submit) or
513              the clean final answer.
514        """
515        cmd = [
516            resolve_cmd("claude"),
517            "--output-format",
518            "stream-json",
519            "--verbose",
520        ]
521
522        if self.continue_conversation:
523            if self._last_session_id:
524                cmd.extend(["--resume", self._last_session_id])
525            elif not is_first:
526                cmd.extend(["--resume", "latest"])
527
528        stream = StreamJsonCommand(cmd, input_text=user_input)
529        result_text = ""
530
531        try:
532            for event in stream:
533                etype = event.get("type")
534
535                if etype == "system" and event.get("subtype") == "init":
536                    self._last_session_id = event.get("session_id")
537
538                elif etype == "assistant":
539                    content_blocks = event.get("message", {}).get("content", [])
540                    for block in content_blocks:
541                        if block.get("type") != "text":
542                            continue
543
544                        raw_text = block.get("text", "")
545
546                        # Fire a tool_start event for each new tool call seen
547                        if self.show_tool_events:
548                            for m in re.finditer(r"<function=(\w+)>", raw_text):
549                                tool_name = m.group(1)
550                                if tool_name not in active_tools:
551                                    active_tools[tool_name] = time.time()
552                                    preceding = raw_text[: m.start()]
553                                    explanation = self.__strip_tool_calls__(
554                                        preceding
555                                    ).strip()
556                                    self.tool_event(
557                                        tool_name,
558                                        "running",
559                                        is_running=True,
560                                        detail=explanation or None,
561                                    )
562
563                        # Only print live on intermediate (tool-call) turns.
564                        # On the final turn the base class prints the return value.
565                        if print_assistant:
566                            clean = self.__strip_tool_calls__(raw_text)
567                            if clean:
568                                self.print_response(self.name, clean)
569
570                elif etype == "result":
571                    result_text = event.get("result", "")
572
573                    if event.get("is_error") or event.get("subtype") == "error":
574                        for tool_name, start in list(active_tools.items()):
575                            if self.show_tool_events:
576                                self.tool_event(
577                                    tool_name,
578                                    "done",
579                                    is_running=False,
580                                    elapsed=time.time() - start,
581                                )
582                        active_tools.clear()
583                        stream.wait()
584                        return False, f"Error: {result_text}"
585        except Exception:
586            stream.kill()
587            raise
588
589        stream.wait()
590
591        if not result_text:
592            stderr_text = "".join(stream.stderr_lines).strip()
593            if stderr_text:
594                return False, f"Error: {stderr_text}"
595
596        # Mirror sdk_worker: if the result contains a raw <tool_call>, re-submit
597        if "</tool_call>" in result_text:
598            return True, result_text
599
600        # Clean final turn — close all open tool events
601        for tool_name, start in list(active_tools.items()):
602            elapsed = time.time() - start
603            if self.show_tool_events:
604                self.tool_event(
605                    tool_name, "done", is_running=False, elapsed=elapsed
606                )
607        active_tools.clear()
608
609        return False, self.__strip_tool_calls__(result_text).strip()

611    def respond(
612        self, user_input: str, is_continuation: bool = False
613    ) -> AgentResponse:
614        """
615        Runs one or more `claude -p` subprocess turns until a clean result is
616        received. Intermediate turns (where tool calls are in flight) print
617        assistant text live. The final answer is parsed into an AgentResponse.
618        """
619        is_first = self.first_call
620        self.first_call = False
621
622        active_tools: dict[str, float] = {}
623
624        prompt = user_input
625        if is_continuation:
626            prompt = "Please continue."
627
628        current_input = self.format_prompt(prompt)
629
630        # Hard cap on <tool_call> re-query turns. If the CLI keeps emitting a
631        # tool call it can't execute (malformed/unsupported), this stops the
632        # voice pipeline from hanging forever on that turn.
633        max_retries = 5
634        retries = 0
635        while True:
636            needs_continuation, result_text = self.__run_turn__(
637                current_input,
638                is_first=is_first,
639                active_tools=active_tools,
640                # Print assistant text live only on continuation turns.
641                # The final turn returns its text; the base class prints it once.
642                print_assistant=False,
643            )
644            is_first = False
645
646            if not needs_continuation:
647                return self.parse_output(result_text)
648
649            retries += 1
650            if retries > max_retries:
651                return self.parse_output(
652                    "Error: Exceeded max tool-call re-query attempts "
653                    f"({max_retries}); last raw response: {result_text}"
654                )
655
656            current_input = result_text
