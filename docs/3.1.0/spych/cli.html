 84    def __init__(self) -> None:
 85        self._thread: threading.Thread | None = None
 86        self._stop_event = threading.Event()
 87        self._message = ""
 88        self._verb_thread: threading.Thread | None = None
 89        self._running = False
 90
 91    def start(self, message: str | None = None) -> None:
 92        # Stop any existing spinner before starting a new one,
 93        # preventing leaked threads from double start() calls.
 94        if self._thread and self._thread.is_alive():
 95            self.stop()
 96        self._running = True
 97        self._stop_event.clear()
 98        if message:
 99            self._message = message
100        self._thread = threading.Thread(target=self._spin, daemon=True)
101        self._thread.start()

103    def start_with_verbs(
104        self,
105        name: str,
106        verbs: list[str] | None = None,
107        interval: float = 10.0,
108    ) -> None:
109        """
110        Start the spinner with a cycling verb message: "<name> is <verb>".
111        The verb rotates through `verbs` every `interval` seconds.
112
113        Requires:
114
115        - `name`:
116            - Type: str
117            - What: The subject displayed before the verb (e.g. "Claude")
118
119        Optional:
120
121        - `verbs`:
122            - Type: list[str] | None
123            - What: Verbs to cycle through. Defaults to CliSpinner.DEFAULT_VERBS
124            - Default: None
125
126        - `interval`:
127            - Type: float
128            - What: Seconds between each verb swap
129            - Default: 10.0
130        """
131        verbs = verbs if verbs is not None else self.DEFAULT_VERBS
132
133        def _get_random_message():
134            idx = random.randrange(len(verbs))
135            return f"{name} is {verbs[idx]}"
136
137        self.start(_get_random_message())
138
139        def _verb_cycle() -> None:
140            while not self._stop_event.wait(timeout=interval):
141                self.update(_get_random_message())
142
143        self._verb_thread = threading.Thread(target=_verb_cycle, daemon=True)
144        self._verb_thread.start()
145
146    def update(self, message: str) -> None:
147        self._message = message
148
149    def stop(self, final_message: str | None = None) -> None:
150        was_running = self._running
151        self._running = False
152        self._stop_event.set()
153        # Guard join() so it's only called when the thread actually ran.
154        if self._thread and self._thread.is_alive():
155            self._thread.join()
156        self._thread = None
157        if self._verb_thread and self._verb_thread.is_alive():
158            self._verb_thread.join()
159        self._verb_thread = None
160        # Clear the spinner line
161        sys.stdout.write("\r\033[2K")
162        sys.stdout.flush()
163        if final_message:
164            print(final_message)
165        return was_running
166
167    def _spin(self) -> None:
168        frame_idx = 0
169        color_idx = 0
170        dot_count = 0
171        while not self._stop_event.is_set():
172            frame = self.FRAMES[frame_idx % len(self.FRAMES)]
173            color = self.COLORS[color_idx % len(self.COLORS)]
174            dots = "." * (dot_count % 4)
175
176            visible_content = f"  {frame}  {self._message}{dots:<3}"
177            padding = max(0, 60 - _visible_len(visible_content)) * " "
178            line = (
179                f"\r  {color}{CliColor.BOLD}{frame}{CliColor.RESET}  "
180                f"{CliColor.WHITE}{self._message}{CliColor.GRAY}{dots:<3}{CliColor.RESET}"
181                f"{padding}"
182            )
183            sys.stdout.write(line)
184            sys.stdout.flush()
185
186            time.sleep(0.08)
187            frame_idx += 1
188            if frame_idx % 5 == 0:
189                dot_count += 1
190            if frame_idx % 20 == 0:
191                color_idx += 1
