 22        """
 23        Usage:
 24
 25        - A responder that pipes transcribed audio into the OpenCode CLI
 26          (`opencode run --format json`) via a subprocess and returns the
 27          final response string. Fires live tool-call events as the subprocess
 28          streams them.
 29
 30        Requires:
 31
 32        - `spych_object`:
 33            - Type: Spych
 34            - What: An initialized Spych instance used to record and transcribe audio
 35
 36        Optional:
 37
 38        - `continue_conversation`:
 39            - Type: bool
 40            - What: Whether to pass `--session` / `--continue` to reuse the most recent session
 41            - Default: True
 42
 43        - `listen_duration`:
 44            - Type: int | float | str
 45            - What: How long to listen for after the wake word is detected
 46            - Default: 0 (Auto detect)
 47            - Options:
 48                - int | float : Record for exactly this many seconds
 49                - "auto" or 0 : Use Silero VAD to detect a complete utterance and
 50                                stop automatically when the speaker finishes
 51
 52        - `name`:
 53            - Type: str
 54            - What: A custom name for the responder to use in printed messages
 55            - Default: "OpenCode"
 56
 57        - `show_tool_events`:
 58            - Type: bool
 59            - What: Whether to print tool start/end events in the CLI as they arrive
 60            - Default: True
 61
 62        - `model`:
 63            - Type: str
 64            - What: Model to use in provider/model format (e.g. "anthropic/claude-sonnet-4-5")
 65            - Default: None (uses opencode default)
 66
 67        - `use_speaker`:
 68            - Type: bool
 69            - What: Whether to speak responses aloud via kokoro TTS after printing them
 70            - Default: False
 71
 72        - `speaker_voice`:
 73            - Type: str
 74            - What: A kokoro voice ID used for all spoken responses
 75            - Default: "af_heart"
 76            - Note: American English voices use prefix `am_` or `af_`; British English
 77              use `bm_` or `bf_`. See spych.speaker.Speaker for the full voice list.
 78
 79        - `response_style`:
 80            - Type: str | None
 81            - What: Style preset or custom instruction shaping how the LLM formats its
 82              summary. Named presets: concise, friendly, military, five_year_old, fast,
 83              pirate, news_anchor, haiku, shakespearean, robot, caveman, yoda, jarvis.
 84              Any other string is used verbatim as a custom instruction.
 85            - Default: None
 86
 87        Notes:
 88
 89        - Uses `opencode run --format json` to stream newline-delimited JSON events
 90        - Event format (observed from real CLI output):
 91            - "step_start"  - model turn beginning; carries sessionID
 92            - "text"        - streaming text delta; part.text accumulates the response.
 93                              May contain inline tool XML: <function=Name>...</function></tool_call>
 94                              These are stripped from the final answer.
 95            - "step_finish" - turn done; part.reason == "stop" signals the final turn.
 96                              Intermediate steps (tool turns) have reason != "stop".
 97        - Tool calls appear as inline XML within "text" events, not as separate event types.
 98          They are extracted for display, then stripped from the final answer text.
 99        - Session continuation uses --session <id> when available, else --continue.
100        - OpenCode CLI must be installed and authenticated before use.
101        """

103        super().__init__(
104            spych_object=spych_object,
105            listen_duration=listen_duration,
106            name=name,
107            use_speaker=use_speaker,
108            speaker_voice=speaker_voice,
109            response_style=response_style,
110            **kwargs,
111        )

113        self.show_tool_events = show_tool_events
114        self.model = model
115        self.first_call = True
116        self._last_session_id: Optional[str] = None
117

119        # <function=ToolName>\n<parameter=x>val</parameter>\n</function>\n</tool_call>

122            re.DOTALL,
123        )

126        """Strip inline tool-call XML from text, return clean prose only."""
127        return self.TOOL_CALL_RE.sub("", text).strip()

129    def respond(self, user_input: str) -> AgentResponse:
130        """
131        Usage:
132
133        - Pipes the transcribed user input into `opencode run --format json`
134          and returns a structured AgentResponse after all tool calls complete.
135          Fires tool events live as they arrive from text deltas.
136
137        Requires:
138
139        - `user_input`:
140            - Type: str
141            - What: The transcribed text from the user's audio input
142
143        Returns:
144
145        - `response`:
146            - Type: AgentResponse
147            - What: Parsed structured response (inline tool XML stripped before parse)
148        """
149        is_first = self.first_call
150        self.first_call = False
151
152        # Build command — prompt goes last as positional arg
153        cmd = ["opencode", "run", "--format", "json"]
154
155        if self.continue_conversation:
156            if self._last_session_id:
157                cmd.extend(["--session", self._last_session_id])
158            elif not is_first:
159                cmd.append("--continue")
160
161        if self.model:
162            cmd.extend(["--model", self.model])
163
164        cmd.append(self.format_prompt(user_input))
165
166        proc = subprocess.Popen(
167            cmd,
168            stdin=subprocess.DEVNULL,
169            stdout=subprocess.PIPE,
170            stderr=subprocess.PIPE,
171            text=True,
172        )
173
174        # Accumulates the full streamed text for the current step.
175        # Resets on each intermediate step_finish (tool turn), kept on final stop.
176        accumulated_text = ""
177
178        # Tracks tool calls seen in the current step: name -> start_time.
179        # Keyed by tool name since inline XML has no unique call ID.
180        active_tools: dict[str, float] = {}
181
182        for raw_line in proc.stdout:
183            raw_line = raw_line.strip()
184            if not raw_line:
185                continue
186
187            try:
188                event = json.loads(raw_line)
189            except json.JSONDecodeError:
190                continue
191
192            etype = event.get("type")
193
194            # Capture session ID from any event for conversation continuity
195            session_id = event.get("sessionID")
196            if session_id:
197                self._last_session_id = session_id
198
199            if etype == "step_start":
200                # New model turn beginning — reset accumulator for this step
201                accumulated_text = ""
202
203            elif etype == "text":
204                delta = event.get("part", {}).get("text", "")
205                if not delta:
206                    continue
207
208                # Check for new tool calls appearing in this delta
209                if self.show_tool_events:
210                    for match in self.TOOL_CALL_RE.finditer(delta):
211                        tool_name = match.group(1)
212                        if tool_name not in active_tools:
213                            # Extract any prose before the tool call as explanation
214                            preceding = delta[: match.start()]
215                            explanation = self.__strip_tool_calls__(
216                                preceding
217                            ).strip()
218                            active_tools[tool_name] = time.time()
219                            self.tool_event(
220                                tool_name, explanation, is_running=True
221                            )
222
223                accumulated_text = delta  # opencode sends full text each delta, not incremental
224
225            elif etype == "step_finish":
226                reason = event.get("part", {}).get("reason", "")
227
228                # Close all open tool events for this step
229                if self.show_tool_events:
230                    for tool_name, start in list(active_tools.items()):
231                        elapsed = time.time() - start
232                        self.tool_event(
233                            tool_name, "done", is_running=False, elapsed=elapsed
234                        )
235                active_tools.clear()
236
237                if reason != "stop":
238                    # Intermediate turn (tool execution step) — reset and continue
239                    accumulated_text = ""
240
241                # On reason == "stop", keep accumulated_text as the final answer
242
243        proc.wait()
244
245        # Close any still-open tool events (e.g. if process ended unexpectedly)
246        for tool_name, start in list(active_tools.items()):
247            elapsed = time.time() - start
248            if self.show_tool_events:
249                self.tool_event(
250                    tool_name, "done", is_running=False, elapsed=elapsed
251                )
252        active_tools.clear()
253
254        return self.parse_output(self.__strip_tool_calls__(accumulated_text))

303        - What: Whether to print tool start/end events in the CLI as they arrive
304        - Default: True

308        - What: Model to use in provider/model format (e.g. "anthropic/claude-sonnet-4-5")

310

312        - Type: str
313        - What: A custom display name for the responder shown in printed messages
314        - Default: None (uses "OpenCode")

317        - Type: bool
318        - What: Whether to speak responses aloud via kokoro TTS

323        - What: Kokoro voice ID for spoken responses
324        - Default: "af_heart"

328        - What: Style preset or custom instruction for the LLM's summary output (e.g. "military", "jarvis")
329        - Default: None
330

332        - Type: dict
333        - What: Additional keyword arguments to pass to the Spych constructor
334        - Default: None
335

337        - Type: dict
338        - What: Additional keyword arguments to pass to SpychWake via SpychOrchestrator
339        - Default: None

136          Fires tool events live as they arrive from text deltas.

139
140        - `user_input`:
141            - Type: str
142            - What: The transcribed text from the user's audio input
143
144        Returns:
145
146        - `response`:
147            - Type: AgentResponse
148            - What: Parsed structured response (inline tool XML stripped before parse)
149        """
150        is_first = self.first_call
151        self.first_call = False
152
153        # Build command — prompt goes last as positional arg
154        cmd = ["opencode", "run", "--format", "json"]
155
156        if self.continue_conversation:
157            if self._last_session_id:
158                cmd.extend(["--session", self._last_session_id])
159            elif not is_first:
160                cmd.append("--continue")
161
162        if self.model:
163            cmd.extend(["--model", self.model])
164
165        cmd.append(self.format_prompt(user_input))
166
167        proc = subprocess.Popen(
168            cmd,
169            stdin=subprocess.DEVNULL,
170            stdout=subprocess.PIPE,
171            stderr=subprocess.PIPE,
172            text=True,
173        )
174
175        # Accumulates the full streamed text for the current step.
176        # Resets on each intermediate step_finish (tool turn), kept on final stop.
177        accumulated_text = ""
178
179        # Tracks tool calls seen in the current step: name -> start_time.
180        # Keyed by tool name since inline XML has no unique call ID.
181        active_tools: dict[str, float] = {}
182
183        for raw_line in proc.stdout:
184            raw_line = raw_line.strip()
185            if not raw_line:
186                continue
187
188            try:
189                event = json.loads(raw_line)
190            except json.JSONDecodeError:
191                continue
192
193            etype = event.get("type")
194
195            # Capture session ID from any event for conversation continuity
196            session_id = event.get("sessionID")
197            if session_id:
198                self._last_session_id = session_id
199
200            if etype == "step_start":
201                # New model turn beginning — reset accumulator for this step
202                accumulated_text = ""
203
204            elif etype == "text":
205                delta = event.get("part", {}).get("text", "")
206                if not delta:
207                    continue
208
209                # Check for new tool calls appearing in this delta
210                if self.show_tool_events:
211                    for match in self.TOOL_CALL_RE.finditer(delta):
212                        tool_name = match.group(1)
213                        if tool_name not in active_tools:
214                            # Extract any prose before the tool call as explanation
215                            preceding = delta[: match.start()]
216                            explanation = self.__strip_tool_calls__(
217                                preceding
218                            ).strip()
219                            active_tools[tool_name] = time.time()
220                            self.tool_event(
221                                tool_name, explanation, is_running=True
222                            )
223
224                accumulated_text = delta  # opencode sends full text each delta, not incremental
225
226            elif etype == "step_finish":
227                reason = event.get("part", {}).get("reason", "")
228
229                # Close all open tool events for this step
230                if self.show_tool_events:
231                    for tool_name, start in list(active_tools.items()):
232                        elapsed = time.time() - start
233                        self.tool_event(
234                            tool_name, "done", is_running=False, elapsed=elapsed
235                        )
236                active_tools.clear()
237
238                if reason != "stop":
239                    # Intermediate turn (tool execution step) — reset and continue
240                    accumulated_text = ""
241
242                # On reason == "stop", keep accumulated_text as the final answer
243
244        proc.wait()
245
246        # Close any still-open tool events (e.g. if process ended unexpectedly)
247        for tool_name, start in list(active_tools.items()):
248            elapsed = time.time() - start
249            if self.show_tool_events:
250                self.tool_event(
251                    tool_name, "done", is_running=False, elapsed=elapsed
252                )
253        active_tools.clear()
254
255        return self.parse_output(self.__strip_tool_calls__(accumulated_text))
