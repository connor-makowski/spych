spych.session_store

Session persistence for spych agent conversations.

Each invocation of a spych agent gets a UUID session ID. The ID is used as the filename under ~/.config/spych/sessions/ and the file records the workspace path plus the underlying agent conversation ID so the session can be resumed across restarts.

 15_SESSIONS_DIR = os.path.join(
 16    os.path.expanduser("~"), ".config", "spych", "sessions"

 20def _sessions_dir() -> str:
 21    """
 22    Usage:
 23
 24    - Return the sessions directory path, creating it if needed.
 25
 26    Returns:
 27
 28    - `path`:
 29        - Type: str
 30        - What: Absolute path to ``~/.config/spych/sessions/``.

 36def _session_path(session_id: str) -> str:
 37    return os.path.join(_sessions_dir(), f"{session_id}.json")

 40def new_session_id() -> str:
 41    """
 42    Usage:

 46    Returns:

 55def load_session(session_id: str) -> dict:
 56    """
 57    Usage:
 58
 59    - Load a previously saved session by its UUID.
 60
 61    Requires:
 62
 63    - `session_id`:
 64        - Type: str
 65        - What: UUID of the session to load.
 66
 67    Returns:
 68
 69    - `data`:
 70        - Type: dict
 71        - What: Session data dict with keys ``session_id``, ``workspace``,
 72          ``conversation_id`` (may be None), and ``created_at``.
 73
 74    Notes:
 75
 76    - Returns an empty dict if the file does not exist or cannot be parsed.
 77    """
 78    path = _session_path(session_id)
 79    try:
 80        with open(path, "r", encoding="utf-8") as fh:
 81            return json.load(fh)
 82    except Exception:
 83        return {}

 86def save_session(
 87    session_id: str, workspace: str, conversation_id: Optional[str]
 88) -> None:
 89    """
 90    Usage:
 91
 92    - Persist a session's workspace and agent conversation ID to disk.
 93
 94    Requires:
 95
 96    - `session_id`:
 97        - Type: str
 98        - What: UUID of this spych session (used as the filename).
 99
100    - `workspace`:
101        - Type: str
102        - What: Absolute path to the workspace directory.
103
104    - `conversation_id`:
105        - Type: str | None
106        - What: The agent's conversation ID to resume on the next run.
107
108    Notes:
109
110    - Overwrites any existing file for the same ``session_id``.
111    """
112    path = _session_path(session_id)
113    existing = load_session(session_id)
114    data = {
115        "session_id": session_id,
116        "workspace": workspace,
117        "conversation_id": conversation_id,
118        "created_at": existing.get("created_at") or _now_iso(),
119        "updated_at": _now_iso(),
120    }
121    with open(path, "w", encoding="utf-8") as fh:
122        json.dump(data, fh, indent=2)

125def latest_session_for_workspace(workspace: str) -> Optional[str]:
126    """
127    Usage:
128
129    - Find the most recently updated session ID for a given workspace.
130
131    Requires:
132
133    - `workspace`:
134        - Type: str
135        - What: Absolute path to the workspace directory.
136
137    Returns:
138
139    - `session_id`:
140        - Type: str | None
141        - What: The UUID of the most recent matching session, or None if no
142          session exists for that workspace.
143    """
144    sessions_dir = _sessions_dir()
145    best_id: Optional[str] = None
146    best_mtime: float = -1.0
147
148    try:
149        entries = os.listdir(sessions_dir)
150    except OSError:
151        return None
152
153    for fname in entries:
154        if not fname.endswith(".json"):
155            continue
156        fpath = os.path.join(sessions_dir, fname)
157        try:
158            mtime = os.path.getmtime(fpath)
159            with open(fpath, "r", encoding="utf-8") as fh:
160                data = json.load(fh)
161            if data.get("workspace") == workspace and mtime > best_mtime:
162                best_mtime = mtime
163                best_id = data.get("session_id")
164        except Exception:
165            continue
166
167    return best_id

170def _now_iso() -> str:
171    from datetime import datetime, timezone
172
173    return datetime.now(timezone.utc).isoformat()
